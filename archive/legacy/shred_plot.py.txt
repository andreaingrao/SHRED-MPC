def numpy_loader(numpy_dir):

    states_vec = np.load(os.path.join(numpy_dir, "states_vec.npy"))

    measurement_ids= np.load(os.path.join(numpy_dir, "measurement_ids.npy"))
    
    points_vec  = np.load(os.path.join(numpy_dir, "points_vec.npy"))

    parameters_vec  = np.load(os.path.join(numpy_dir, "parameters_vec.npy"))

    def _load_npz(fname):
        with np.load(fname, allow_pickle=False) as data:
            return {k: np.array(v) for k, v in data.items()}

    recon = _load_npz(os.path.join(NUMPY_DIRECTORY, "recon.npz"))

    return (states_vec, measurement_ids, points_vec,
            parameters_vec, recon)

print(f"\n----------------------\nLoading NumPy Data ...\n----------------------\n")
