states_vec, points_vec, parameters_time_tensor      = data_load("case_omega") # parameters_time_tensor: [cases, timesteps, parameter]

#### SHRED TRAINING ####

manager_pod, model, parameters_time_tensor_from_training = shred_training(states_vec = states_vec, measurements_vec = measurements_vec_training, parameters_time_tensor=parameters_time_tensor)

engine_pod                                          = ParametricSHREDEngine(manager_pod, model)



#### MPC CYCLE ####

t = 0

x = points_vec[..., 0]
y = points_vec[..., 1]

x_meas = x[sensor_ids]
y_meas = y[sensor_ids]

cwd = os.getcwd()

ux_zero = np.zeros((meas_offset, x.shape[-1]), dtype=states_vec.dtype)
uy_zero = np.zeros((meas_offset, y.shape[-1]), dtype=states_vec.dtype)

print(ux_zero.shape)

omegas = []

for d in range(0, meas_offset):

    omegas.append(0)

while t < meas_offset * N_cycles:

    os.chdir(cwd)

    ### CREATE MEASUREMENTS FROM LIVE DATA ###
    measurements = sensor_maker(states_vec[-1, :, :])
    
    print("measurements")
    print(measurements.shape)
    measurements_flat = measurements.reshape(measurements.shape[1], -1)

    print("measurements_flat")
    print(measurements_flat.shape)


    ### DECODE ALL POSSIBLE PARAMETERS ###
    delta_max = 2
    sensors_mpc, parameters_distributed = sensors_equal_distribution(measurements_flat=measurements_flat, delta_max= delta_max, N=30)
    
    print(sensors_mpc.shape)
    ux_pred, uy_pred = decode_distributed_states(sensors=sensors_mpc, parameters_distributed=parameters_distributed, engine_pod=engine_pod)


    ### EVALUATE LOSS FUNCTION ###
    # NOTA: predictions sono sensate nell'intorno di omega0, smettono di esserlo allontanandosi troppo
    state_pred = np.stack([ux_pred, uy_pred], axis= -1)
    loss_vec = loss_vec_generator(state=state_pred)

    # plt.plot(parameters_distributed, loss_vec, 'ro')
    # plt.show()


    ### MINIMIZE LOSS FUNCTION ###
    omega_min = np.argmin(loss_vec)
    par_optimum = float(parameters_distributed[omega_min])
    loss_min_pred = loss_vec[omega_min]


    ### SIMULATE FURTHER ###
    omegas = OpenFoam_launcher(par_optimum=par_optimum, omegas=omegas)

    mag_x, mag_y = mag_cfd_extractor(omegas=omegas)

    mag_true = np.sqrt(mag_x**2 + mag_y**2)

    states_vec = np.stack([mag_x, mag_y], axis=-1)

    meas_checks = sensor_maker(states_vec = states_vec[:-meas_offset, :])
    meas_checks_flat = meas_checks.reshape(meas_checks.shape[0], -1)
    
    omega_col = np.asarray(omegas[:-meas_offset]).reshape(-1, 1)
    sensor_checks = np.concatenate([meas_checks_flat, omega_col], axis=1)

    timesteps = np.linspace(0, t + meas_offset, t + meas_offset + 1)

    latent = engine_pod.sensor_to_latent(sensor_checks)

    recon = engine_pod.decode(latent)
    ux_pred = np.asarray(recon["States_x"])
    uy_pred = np.asarray(recon["States_y"])

    ux_pred = np.concatenate([ux_zero, ux_pred], axis = 0)
    uy_pred = np.concatenate([uy_zero, uy_pred], axis = 0)

    state_pred = np.stack([ux_pred, uy_pred], axis = -1)
    mag_pred = np.sqrt(ux_pred**2 + uy_pred**2)


    loss_cfd = energy_dissipated(state = states_vec[:t + meas_offset, ...])
    loss_min_pred = energy_dissipated(state = state_pred)
    

    t = t + meas_offset

output_path = "shred_images"
