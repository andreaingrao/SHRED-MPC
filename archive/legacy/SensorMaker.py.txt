import os
import pandas as pd
import torch

# Path to the root directory containing case_omega folders
root_dir = "DataSet/CSV_files"

all_cases = sorted([d for d in os.listdir(root_dir) if d.startswith("case_omega")])

# Define the cellIDs you want to extract for sensors
sensor_cellIDs = [3950, 3345, 7300] 

states_list = []
sensors_list = []

for case in all_cases:
    case_path = os.path.join(root_dir, case)

    # Collect CSV files (datacsv_x.csv) and sort by timestep number
    csv_files = sorted([f for f in os.listdir(case_path) if f.startswith("datacsv_") and f.endswith(".csv")],
                       key=lambda x: int(x.split("_")[-1].split(".")[0]))

    timestep_states = []
    timestep_sensors = []
    for csv_file in csv_files:
        csv_path = os.path.join(case_path, csv_file)

        # Load CSV
        df = pd.read_csv(csv_path)

        # Extract U:0 and U:1 for all rows
        state_vector = df[["U:0", "U:1"]].to_numpy().flatten()
        timestep_states.append(state_vector)

        # Extract only the rows with desired cellIDs for sensors
        sensor_df = df[df["cellID"].isin(sensor_cellIDs)]
        sensor_vector = sensor_df[["U:0", "U:1"]].to_numpy().flatten()
        timestep_sensors.append(sensor_vector)

    # Stack timesteps for this case
    timestep_states = torch.tensor(timestep_states, dtype=torch.float32)
    timestep_sensors = torch.tensor(timestep_sensors, dtype=torch.float32)

    states_list.append(timestep_states)
    sensors_list.append(timestep_sensors)

# Final tensors
states = torch.stack(states_list)      # [nparameters, ntimesteps, nstate]
sensors = torch.stack(sensors_list)    # [nparameters, ntimesteps, nsensor_state]

print(f"states shape: {states.shape}")
print(f"sensors shape: {sensors.shape}")
