from shred_utils.global_var import *
from shred_utils.input_var import * 
from shred_utils.functions import *
import matplotlib.pyplot as plt



#### INITIAL LOADS ####

engine_pod, states_vec, points_vec, manager_pod = DataLoad_Training("case_omega") 

# estraggo le coordinate di tutti i punti points_vec = [n_points, 2]

x = points_vec[..., 0]
y = points_vec[..., 1]


# estraggo le coordinate dei sensori (for graphic purposes) x, y = [n_points]

x_meas = x[sensor_ids]
y_meas = y[sensor_ids]


# estraggo le ultime misurazioni (per poter prevedere il primo stato successivo al mio attuale
# timestep, visto che faccio previsioni meas_offset più avanti, devo avere un lag di dati indietro di un meas_offset)

measurements = []
parameters_vec   = []
parameters_cfd_sequence = []

lift_pred = []
lift_real = []


parameters_vec.append(omega0)

# mi costruisco anche un lag di omega0, a cui aggiungerò un meas_offset di omega nuovi, visto che i parametri sono
# temporalmente allineati alle ricostruzioni e NON alle misurazioni
#  
for _ in range(shred_lags):
    parameters_cfd_sequence.append(omega0)

cwd = os.getcwd()

t = shred_lags  

if CONTROL_HORIZON <= 0 or CONTROL_HORIZON > meas_delta_t:
    raise ValueError(f"CONTROL_HORIZON must be in [1, {meas_delta_t}], got {CONTROL_HORIZON}")

os.makedirs("shred_images", exist_ok=True)

#### MPC CYCLE ####

while len(lift_real) < N_cycles:


    os.chdir(cwd)

    measurements = states_vec[-(shred_lags + meas_delta_t) : , sensor_ids, ...]  # (T, Ns, 2)
    meas_x = measurements[..., 0]
    meas_y = measurements[..., 1]

    meas_flat = np.concatenate([meas_x, meas_y], axis=-1)
    
    print(f"measurements_flat = {measurements}")

    lift_vec = []
    ux_future_candidates = []
    uy_future_candidates = []
    parameter_future_candidates = []
    
    #### LIFT PREDICTIVE EVALUATION ####
    possible_omegas = [
        parameters_vec[-1] - 0.1*PARAMETER_DELTA,
        parameters_vec[-1],
        parameters_vec[-1] + 0.1*PARAMETER_DELTA,
    ]

    dir = t + CONTROL_HORIZON

    os.makedirs(f"shred_images/{dir}", exist_ok=True)
    
    # nota che qui effettuo la ricostruzione dello spazio, e i miei measurements non cambiano mai! Tengo sempre il meas_flat fuori, concatenandolo solo ai nuovi parametri
    for i in range (len(possible_omegas)):

        omega = float(possible_omegas[i])
                
        omega_apply = np.linspace(parameters_cfd_sequence[-1], omega, CONTROL_HORIZON + 1)[1:]
        omega_hold = np.full(meas_delta_t - CONTROL_HORIZON, omega)
        omega_future = np.concatenate([omega_apply, omega_hold])
        omega_vec = np.concatenate([parameters_cfd_sequence[-shred_lags:], omega_future])


        print(f"MEAS_DIM = {measurements.shape}, OMEGA_DIM = {omega_vec.shape}")
        sensors = np.concatenate([meas_flat, omega_vec[:, None]], axis=-1)[None, ...]

        print(f"SENSORS_DIM = {sensors.shape}")
        
        latent = engine_pod.sensor_to_latent(sensors)
        recon = engine_pod.decode(latent)

        ux_pred = np.asarray(recon["States_x"])
        uy_pred = np.asarray(recon["States_y"])
        par_pred = np.asarray(recon["Parameter"])
        if ux_pred.ndim > 2 and ux_pred.shape[0] == 1:
            ux_pred = ux_pred[0]
        if uy_pred.ndim > 2 and uy_pred.shape[0] == 1:
            uy_pred = uy_pred[0]
        if par_pred.ndim > 2 and par_pred.shape[0] == 1:
            par_pred = par_pred[0]

        print(f" ux_pred {ux_pred.shape}")

        ux_future = ux_pred[-meas_delta_t:, ...]
        uy_future = uy_pred[-meas_delta_t:, ...]
        parameter_future = par_pred[-meas_delta_t:, ...]
        ux_future_candidates.append(ux_future)
        uy_future_candidates.append(uy_future)
        parameter_future_candidates.append(parameter_future)

        # for j in range (meas_delta_t):

        #     os.makedirs(f"shred_images/{dir}/{t + j}", exist_ok=True)
        #     shred_plot(x, y, x_meas, y_meas, mag_pred[-meas_delta_t + j], 0, vmax, omega_line[-meas_delta_t + j], t + j, par_pred[-meas_delta_t + j], 'save', dir)

        ux_last = ux_future[:CONTROL_HORIZON, ...]
        uy_last = uy_future[:CONTROL_HORIZON, ...]

        lift = lift_generated(ux_last, uy_last) / CONTROL_HORIZON

        lift_vec.append(lift)
    

    plt.plot(possible_omegas, lift_vec, 'ro')
    plt.savefig(f"shred_images/{t}.png", dpi=200)
    plt.close('all')
    

    print(f"Lift vec: {lift_vec}")

    lift_vec = np.asarray(lift_vec)
    id_omega_choice= np.argmin((lift_vec - LIFT_TARGET)**2)
    ux_chosen_future = ux_future_candidates[id_omega_choice]
    uy_chosen_future = uy_future_candidates[id_omega_choice]
    parameter_chosen_future = parameter_future_candidates[id_omega_choice]

    lift_pred.append(lift_vec[id_omega_choice])

    parameters_vec.append(possible_omegas[id_omega_choice])

    print(f"VALMIN = {possible_omegas[id_omega_choice]}")


    #### COMPARATIVE PLOT CREATION #### 

    parameters_cfd_sequence = OpenFoam_launcher(parameters_cfd_sequence, parameters_vec[-1], control_horizon=CONTROL_HORIZON)
    os.chdir(cwd)  # return to project root so saves land under the expected folder

    mag_x_cfd, mag_y_cfd, states_vec_cfd = mag_cfd_extractor(count=CONTROL_HORIZON)
    mag_x_cfd_new = mag_x_cfd
    mag_y_cfd_new = mag_y_cfd
    parameter_cfd_new = parameters_cfd_sequence[-CONTROL_HORIZON:]
    timestep_new = len(parameters_cfd_sequence) - CONTROL_HORIZON

    print(f"states_vec: {states_vec.shape}, states_vec_cfd: {states_vec_cfd.shape}")
    states_vec = np.concatenate([states_vec, states_vec_cfd], axis = 0)
    
    lift_cfd = lift_generated(mag_x_cfd_new, mag_y_cfd_new) / CONTROL_HORIZON

    lift_real.append(lift_cfd)

    print(f"STATES_VEC = = = = = = : {states_vec.shape}")
   
    vmax = 0.8

    print(len(parameters_vec))

    for i in range(CONTROL_HORIZON):

        shred_lift_comp_plot(
            x=x,
            y=y,
            x_meas=x_meas,
            y_meas=y_meas,
            mag_x_pred=ux_chosen_future[i, :],
            mag_y_pred=uy_chosen_future[i, :],
            mag_x_true=mag_x_cfd_new[i, :],
            mag_y_true=mag_y_cfd_new[i, :],
            vmax=vmax,
            par_true=parameter_cfd_new[i],
            par_pred=parameter_chosen_future[i],
            timestep=timestep_new + i,
            desired_lift=LIFT_TARGET,
            type='save',
        )


    t = timestep_new + CONTROL_HORIZON


lift_time_plot(lift_vec_cfd=lift_real, lift_vec_pred=lift_pred, desired_lift=LIFT_TARGET, type='save')
