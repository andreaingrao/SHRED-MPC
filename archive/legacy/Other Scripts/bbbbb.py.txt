from shred_utils.global_var import *
from shred_utils.input_var import * 
from shred_utils.functions import *
import matplotlib.pyplot as plt
import shutil


#### INITIAL LOADS ####

engine_pod, state_0, points_vec = DataLoad_Training("case_omega") 
# state_0 first meas_delta_t states (they don't change in any case)

x = tensor_spatial_component(points_vec, 0)
y = tensor_spatial_component(points_vec, 1)

x_meas = x[sensor_ids]
y_meas = y[sensor_ids]


states_vec = state_0

print(f"states_vec: {states_vec.shape}")
measurements = []
parameters   = []

for _ in range(shred_lags):
    parameters.append(omega0)

parameters = np.asarray(parameters, dtype=float)
t = shred_lags 

cwd = os.getcwd()


#### MPC CYCLE ####

while t < shred_lags + meas_offset * N_cycles:

    os.chdir(cwd)

    measurements = states_vec[-shred_lags:, sensor_ids, ...]
    measurements = np.concatenate([measurements[..., 0], measurements[..., 1]], axis=-1)

    print(f"measurements_flat = {measurements}")

    loss_vec = []
    candidate_forecasts = {}

    #### LOSS PREDICTIVE EVALUATION ####
       #### LOSS PREDICTIVE EVALUATION ####
    possible_omegas = [
        parameters[-1] - PARAMETER_DELTA,
        parameters[-1],
        parameters[-1] + PARAMETER_DELTA,
    ]

    for omega in possible_omegas:
        # keep the real history, overwrite only the control we’re about to apply
        omega_hist = np.array(parameters[-shred_lags:], dtype=float)
        omega_hist[-1] = omega

        sensors = np.concatenate([measurements, omega_hist[:, None]], axis=1)[None, ...]

        latent = engine_pod.sensor_to_latent(sensors)
        recon = engine_pod.decode(latent)

        ux_pred = np.asarray(recon["States_x"])
        uy_pred = np.asarray(recon["States_y"])

        ux_window = ux_pred[-1]          # (lags, n_points)
        uy_window = uy_pred[-1]

        mag_last = np.sqrt(ux_window[-1]**2 + uy_window[-1]**2)
        vmax = np.max(mag_last)
        shred_plot(x, y, x_meas, y_meas, mag_last, 0, vmax, omega, t, 'save')

        loss = energy_dissipated(ux_window[-1], uy_window[-1])
        loss_vec.append(loss)

    plt.plot(possible_omegas, loss_vec, 'ro')
    plt.savefig(f"shred_images/{t}.png", dpi=200)
    plt.close()

    print(f"Loss vec: {loss_vec}")

    loss_vec = np.asarray(loss_vec)
    omega = np.argmax(loss_vec)
    best_omega = possible_omegas[omega]

    print(f"VALMAX = {best_omega}")

    best_forecast_window = candidate_forecasts[best_omega]

    #### COMPARATIVE PLOT CREATION #### 

    parameters = OpenFoam_launcher(parameters, best_omega)

    mag_x_cfd, mag_y_cfd, states_vec = mag_cfd_extractor(parameters)
    mag_cfd = np.sqrt(mag_x_cfd**2 + mag_y_cfd**2)

    horizon = best_forecast_window.shape[0]
    cfd_window = mag_cfd[t : t + horizon]
    vmax = max(np.max(best_forecast_window), np.max(cfd_window))

    for i in range(horizon):
        timestep = t + i
        shred_comp_plot(
            x=x,
            y=y,
            x_meas=x_meas,
            y_meas=y_meas,
            mag_pred=best_forecast_window[i],
            mag_true=cfd_window[i],
            vmax=vmax,
            parameter=parameters[timestep],
            timestep=timestep,
            loss_min_pred=0,
            loss_cfd=0,
            type='save'
        )

    t += meas_offset
