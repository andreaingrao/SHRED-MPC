#!/bin/bash
# ==============================
# OpenFOAM Parallel Run Script with safe .foam creation
# ==============================

# Number of processors
NP=16   # change if you want fewer/more cores

# Stop on first error
set -e

# Get the current case name and full path
CASE_NAME=$(basename "$PWD")
CASE_PATH=$(pwd)

echo "----------------------------------"
echo " Running blockMesh..."
echo "----------------------------------"
blockMesh

echo "----------------------------------"
echo " Decomposing domain for $NP cores..."
echo "----------------------------------"
decomposePar -force

echo "----------------------------------"
echo " Running solver in parallel..."
echo "----------------------------------"
mpirun -np $NP simpleFoam -parallel > log.simpleFoam

echo "----------------------------------"
echo " Reconstructing case..."
echo "----------------------------------"
reconstructPar

echo "----------------------------------"
echo " Creating or overwriting .foam file for ParaView..."
echo "----------------------------------"
FOAM_FILE="${CASE_PATH}/${CASE_NAME}.foam"

# Create or overwrite the .foam file
echo "// ParaView .foam file for case ${CASE_NAME}" > "$FOAM_FILE"

echo "Created .foam file at: $FOAM_FILE"

echo "----------------------------------"
echo " Launching ParaView..."
echo "----------------------------------"
paraFoam
