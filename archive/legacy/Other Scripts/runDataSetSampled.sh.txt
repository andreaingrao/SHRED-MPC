#!/bin/bash
# ==============================
# Automated Multi-Omega Runner (post-solver CSV extraction)
# ==============================

# Number of processors
NP=16   # change if needed

# Stop on first error
set -e

BASE_CASE=$(pwd)
DATASET_DIR="${BASE_CASE}/DataSet"
CSVSET_DIR="${BASE_CASE}/DataSetCSV"

mkdir -p "$DATASET_DIR" "$CSVSET_DIR"

# Generate 50 omega values from -10 to 10
OMEGAS=($(seq -10 10 10))   
TOTAL=${#OMEGAS[@]}
COUNT=0

for OMEGA in "${OMEGAS[@]}"; do
    COUNT=$((COUNT + 1))
	OMEGA_FMT=$(printf "%.3f" "$OMEGA" | tr '.' '_')
	CASE_DIR="${DATASET_DIR}/case_omega${OMEGA_FMT}"
	CSV_DIR="${CSVSET_DIR}/case_omega${OMEGA_FMT}"


    echo "========================================="
    echo " Case ${COUNT}/${TOTAL} | omega = ${OMEGA_FMT}"
    echo " Results in ${CASE_DIR}"
    echo " CSVs in ${CSV_DIR}"
    echo "========================================="

    # Copy base case
    rsync -a --exclude="case_omega*" --exclude="DataSet*" "$BASE_CASE/" "$CASE_DIR/"

    cd "$CASE_DIR"
    sed -i "s/\(omega[[:space:]]*\)[0-9.\-]*/\1${OMEGA_FMT}/" 0/U

    echo "Running blockMesh..."
    blockMesh

    echo "Generating global cell centres..."
    postProcess -func writeCellCentres

    echo "Decomposing domain..."
    decomposePar -force

    echo "Running solver in parallel..."
    mpirun -np "$NP" simpleFoam -parallel > log.simpleFoam 2>&1

    echo "Extracting velocity CSVs..."
    mkdir -p "$CSV_DIR"
	
	mkdir -p "$CSV_DIR"
	for T in processor0/[0-9]*; do
	    [ -d "$T" ] || continue   
	    TNAME=$(basename "$T" | tr '.' '_')
	    CSV_FILE="${CSV_DIR}/vel_t${TNAME}.csv"

	    echo "x,y,Ux,Uy" > "$CSV_FILE"

	    for P in processor*; do
		COORD_FILE="$CASE_DIR/$P/0/C.gz"
		VEL_FILE="$CASE_DIR/$P/$TNAME/U.gz"
		if [[ -f "$COORD_FILE" && -f "$VEL_FILE" ]]; then
		    zcat "$COORD_FILE" | awk '/^ *\(/,/^\)/' | tr -d '();' > coords.tmp
		    zcat "$VEL_FILE"   | awk '/^ *\(/,/^\)/' | tr -d '();' > veloc.tmp
		    paste coords.tmp veloc.tmp | awk '{print $1","$2","$4","$5}' >> "$CSV_FILE"
		    rm -f coords.tmp veloc.tmp
		fi
	    done
	done


    echo "Cleaning processor directories..."
    rm -rf processor*

    echo "Finished case ${COUNT}/${TOTAL} (omega = ${OMEGA_FMT})"
    echo
    cd "$BASE_CASE"
done
