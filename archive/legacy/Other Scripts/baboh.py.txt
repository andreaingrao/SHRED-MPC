def extract_measurement_window(states_vec: np.ndarray) -> tuple[np.ndarray, int]:
    """
    Returns the sensor window used for SHRED along with the starting
    time index of that window in the global timeline.
    """
    required_history = shred_lags + meas_offset
    if states_vec.shape[0] < required_history:
        raise RuntimeError(
            f"Need at least {required_history} CFD snapshots; "
            f"only {states_vec.shape[0]} available."
        )

    start_idx = states_vec.shape[0] - required_history
    end_idx = states_vec.shape[0] - meas_offset

    measurements = states_vec[start_idx:end_idx, sensor_ids, :]
    measurements = measurements.reshape(shred_lags, -1)

    return measurements, start_idx


def compose_sensor_tensor(
    measurements: np.ndarray,
    start_idx: int,
    parameters: np.ndarray,
    candidate: float | None = None,
) -> tuple[np.ndarray, np.ndarray]:
    """
    Concatenates the measurements with the parameter history that is
    shifted forward by meas_delta_t. When candidate is provided, it
    is used to fill any parameter entries that extend beyond the
    current parameter history.
    """
    omega_hist = []
    for lag_idx in range(shred_lags):
        param_idx = start_idx + lag_idx + meas_offset
        if param_idx < len(parameters):
            omega_hist.append(parameters[param_idx])
        elif candidate is not None:
            omega_hist.append(float(candidate))
        else:
            raise RuntimeError(
                "Parameter history shorter than required and no candidate value provided."
            )

    omega_hist = np.asarray(omega_hist, dtype=float)
    sensors = np.concatenate([measurements, omega_hist[:, None]], axis=1)[None, ...]
    return sensors, omega_hist
