
from pathlib import Path


import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT))

from shred_utils.input_var import (
    OPENFOAM_DELTA_T,
    OPENFOAM_END_TIME,
    OPENFOAM_WRITE_INTERVAL,
)


CONTROL_DICT = ROOT / "OpenFoam/system/controlDict"

def update_control_dict():
    replacements = {
        "deltaT": OPENFOAM_DELTA_T,
        "endTime": OPENFOAM_END_TIME,
        "writeInterval": OPENFOAM_WRITE_INTERVAL,
    }
    lines = CONTROL_DICT.read_text().splitlines()
    for idx, line in enumerate(lines):
        stripped = line.strip()
        for key, value in replacements.items():
            if stripped.startswith(f"{key}"):
                indent = line[: len(line) - len(stripped)]
                lines[idx] = f"{indent}{key:<14}{value};"
    CONTROL_DICT.write_text("\n".join(lines) + "\n")

if __name__ == "__main__":
    update_control_dict()
