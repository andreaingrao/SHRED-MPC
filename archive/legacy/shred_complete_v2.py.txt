import os, re
import numpy as np
import pandas as pd
from pyshred import ParametricDataManager, SHRED
import warnings
warnings.filterwarnings("ignore")
from pyshred import ParametricSHREDEngine
import matplotlib.pyplot as plt


DATA_DIRECTORY  = "CSV_files"
PARAMETERS_MIN       = -10.0
PARAMETERS_MAX       = 10.0

def num_from_text(s, pattern=r"(\d+)"):
    m = re.search(pattern, s)
    return int(m.group(1))

case_names      = []
for case in os.listdir(DATA_DIRECTORY):
    if case.startswith("case_omega"):
        case_names.append(case)

case_names.sort(key = lambda s: num_from_text(s))
            
states_vec      = []
parameters_vec  = []
points_vec      = None

for case in case_names:

    case_dir = os.path.join(DATA_DIRECTORY, case)
    files_list       = []
    
    for file in os.listdir(case_dir):
        if file.startswith("datacsv_") and file.endswith(".csv"):
            files_list.append(file)
    
    files_list.sort(key = lambda s: num_from_text(s, r"datacsv_(\d+)\.csv"))
    
    states_list  = []

    for f in files_list:
        df = pd.read_csv(os.path.join(case_dir, f))[["U:0","U:1","Points:0","Points:1"]]  
        state = df[["U:0", "U:1"]].to_numpy(dtype=np.float32) # cioè: per ogni timestep mi prendo il vettore delle velocità in ogni punto
        
        states_list.append(state) # poi quel tensore lo aggiungo a tutti gli altri di ogni timestep (e così creo lista con ogni punto per ogni timestep) 
        
        if points_vec is None:
            points_vec = df[["Points:0", "Points:1"]].to_numpy()
            
    states_vec.append(np.stack(states_list, axis = 0)) #a questo punto rendo la lista che ho creato precedentemente un tensore di tutte le velocità per il timestep, e poi la rendo una lista per tutti i casi con ogni elemento che è il tensore di tutti i vettori di tutti i timestep
    
    case_id         = int(case.replace("case_omega", ""))
    parameter_val = PARAMETERS_MIN + (PARAMETERS_MAX - PARAMETERS_MIN) * (case_id - 1) / (len(case_names) - 1)
    parameters_vec.append(parameter_val)
    
states_vec = np.stack(states_vec, axis=0)
parameters_vec = np.array(parameters_vec, dtype=np.float32)

states_x = states_vec[..., 0]
states_y = states_vec[..., 1]

n_cases, nt, npts = states_x.shape

parameters_time_tensor = np.repeat(parameters_vec[:, None, None], nt, axis=1).astype(np.float32)


manager_pod = ParametricDataManager(
    lags=25,        
    train_size=0.8,
    val_size=0.1,
    test_size=0.1,
)

manager_pod.add_data(
    data=states_x,   
    id="States_x",
    random=5,        
    compress=32
)

manager_pod.add_data(
    data=states_y,
    id="States_y",
    compress=32
)

manager_pod.add_data(
    data=parameters_time_tensor,  
    id="Parameter",
    compress=False
)

train_dataset, val_dataset, test_dataset = manager_pod.prepare()

model = SHRED(
    sequence_model="LSTM",
    decoder_model="MLP",
    latent_forecaster=None
)

val_errors = model.fit(
    train_dataset=train_dataset,
    val_dataset=val_dataset,
    num_epochs=50,
    patience=70,
    verbose=True
)

print("Train MSE:", model.evaluate(train_dataset))
print("Val   MSE:", model.evaluate(val_dataset))
print("Test  MSE:", model.evaluate(test_dataset))

engine_pod = ParametricSHREDEngine(manager_pod, model)

test_latent = engine_pod.sensor_to_latent(manager_pod.test_sensor_measurements)
test_recon  = engine_pod.decode(test_latent)

train_latent = engine_pod.sensor_to_latent(manager_pod.train_sensor_measurements)
train_recon  = engine_pod.decode(train_latent)

val_latent = engine_pod.sensor_to_latent(manager_pod.val_sensor_measurements)
val_recon  = engine_pod.decode(val_latent)

# sizes
nt    = states_x.shape[1]
npts  = states_x.shape[2]
ntrain = len(manager_pod.train_indices)
nval   = len(manager_pod.val_indices)
ntest  = len(manager_pod.test_indices)

def _reshape_field(arr, nsplit, nt, npts, name):
    arr = np.asarray(arr)
    if arr.ndim == 2 and arr.shape == (nsplit*nt, npts):
        return arr.reshape(nsplit, nt, npts)
    if arr.ndim == 3 and arr.shape == (nsplit, nt, npts):
        return arr
    raise ValueError(f"{name} has unexpected shape {arr.shape} (expected {(nsplit*nt, npts)} or {(nsplit, nt, npts)})")



# reshape for each split and each field id you used
train_recon["States_x"] = _reshape_field(train_recon["States_x"], ntrain, nt, npts, "States_x(train)")
train_recon["States_y"] = _reshape_field(train_recon["States_y"], ntrain, nt, npts, "States_y(train)")

val_recon["States_x"]   = _reshape_field(val_recon["States_x"], nval, nt, npts, "States_x(val)")
val_recon["States_y"]   = _reshape_field(val_recon["States_y"], nval, nt, npts, "States_y(val)")

test_recon["States_x"]  = _reshape_field(test_recon["States_x"], ntest, nt, npts, "States_x(test)")
test_recon["States_y"]  = _reshape_field(test_recon["States_y"], ntest, nt, npts, "States_y(test)")


train_case_indices = manager_pod.train_indices
val_case_indices   = manager_pod.val_indices
test_case_indices  = manager_pod.test_indices

train_omegas = parameters_vec[train_case_indices]
val_omegas   = parameters_vec[val_case_indices]
test_omegas  = parameters_vec[test_case_indices]

def plot_split(split: str, which_case: int, which_time: int, cmap="rainbow", size=35):
    if split == "train":
        recon = train_recon; indices = train_case_indices; omegas = train_omegas
    elif split == "val":
        recon = val_recon; indices = val_case_indices; omegas = val_omegas
    elif split == "test":
        recon = test_recon; indices = test_case_indices; omegas = test_omegas
    else:
        raise ValueError("split must be 'train', 'val', or 'test'")

    print("indices", indices)

    case_idx = int(indices[which_case])

    # ground truth
    ux_true = np.asarray(states_x[case_idx, which_time]).reshape(-1)
    uy_true = np.asarray(states_y[case_idx, which_time]).reshape(-1)

    # reconstruction (already in split order)
    ux_pred = np.asarray(recon["States_x"][which_case, which_time]).reshape(-1)
    uy_pred = np.asarray(recon["States_y"][which_case, which_time]).reshape(-1)

    x = points_vec[:, 0].astype(float)
    y = points_vec[:, 1].astype(float)

    mag_true = np.sqrt(ux_true**2 + uy_true**2)
    mag_pred = np.sqrt(ux_pred**2 + uy_pred**2)
    mag_delta= np.abs(mag_pred-mag_true)

    vmin = float(min(mag_true.min(), mag_pred.min()))
    vmax = float(max(mag_true.max(), mag_pred.max()))
    
    # err_max   = float(max(mag_delta)) / 6
    err_max = 1

    # create figure and axes
    fig, axs = plt.subplots(3, 1, figsize=(12, 9), constrained_layout=True)

    # CFD plot
    ax = axs[0]
    scatt0 = ax.scatter(x, y, c=mag_true, cmap=cmap, s=size,
                       edgecolor="none", vmin=vmin, vmax=vmax)
    ax.set_aspect("equal")
    ax.axis("off")
    ax.set_title(f"CFD result, omega = {omegas[which_case]:.3f}, time = {which_time}")

    # SHRED recon plot
    ax = axs[1]
    scatt1 = ax.scatter(x, y, c=mag_pred, cmap=cmap, s=size,
                       edgecolor="none", vmin=vmin, vmax=vmax)
    ax.set_aspect("equal")
    ax.axis("off")
    ax.set_title(f"SHRED reconstruction, omega = {omegas[which_case]:.3f}, time = {which_time}")

    fig.colorbar(mappable=scatt1, ax=axs[1], label="|U|", orientation='horizontal')

    # error
    ax = axs[2]
    scatt2 = ax.scatter(x, y, c=mag_delta, cmap="Greys", s=size,
                       edgecolor="none", vmin=0, vmax=err_max)
    ax.set_aspect("equal")
    ax.axis("off")
    ax.set_title(f"Error Magnitude, max error= {err_max}")


# visualize a TRAIN case
plot_split("train", which_case=0, which_time=50)
plt.savefig("shred_images/train.png", dpi=1000)

# visualize a VAL case
plot_split("val", which_case=0, which_time=50)
plt.savefig("shred_images/val.png", dpi=1000)

# visualize a TEST case 
plot_split("test", which_case=0, which_time=50)
plt.savefig("shred_images/test.png", dpi=1000)
