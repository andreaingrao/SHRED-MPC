"""Regenerate SHRED statistical plots from a completed candidate CSV.

Usage:
    python3 plot_shred_statistics.py
    python3 plot_shred_statistics.py path/to/candidate_statistics.csv

With no path, the cross-training aggregate CSV is preferred.  Passing a
specific run CSV produces plots for only that trained model.
"""

import argparse
import csv
from pathlib import Path

import matplotlib

matplotlib.use("Agg")
import matplotlib.pyplot as plt
from matplotlib.ticker import PercentFormatter
import numpy as np


OMEGA_DECIMALS = 6
REQUIRED_COLUMNS = {
    "eligible_for_statistics",
    "delta_omega",
    "outgoing_omega",
    "pressure_relative_l2_error",
    "lift_relative_l2_error",
}


def default_candidate_csv(root=Path("shred_stat_results")):
    aggregate = root / "candidate_statistics_all_runs.csv"
    if aggregate.exists():
        return aggregate
    candidates = sorted(root.glob("run_*/candidate_statistics.csv"))
    if not candidates:
        raise FileNotFoundError(
            f"No aggregate or run_*/candidate_statistics.csv files found under {root}"
        )
    return candidates[-1]


def load_candidate_statistics(csv_path, include_warmup=False):
    csv_path = Path(csv_path)
    with csv_path.open(newline="") as csv_file:
        reader = csv.DictReader(csv_file)
        missing = REQUIRED_COLUMNS.difference(reader.fieldnames or [])
        if missing:
            raise ValueError(f"Missing required CSV columns: {sorted(missing)}")

        rows = [
            row
            for row in reader
            if include_warmup or int(row["eligible_for_statistics"]) == 1
        ]

    if not rows:
        raise ValueError(f"No candidate rows selected from {csv_path}")

    return {
        "delta_omega": np.asarray(
            [float(row["delta_omega"]) for row in rows],
            dtype=float,
        ),
        "outgoing_omega": np.asarray(
            [float(row["outgoing_omega"]) for row in rows],
            dtype=float,
        ),
        "pressure_error": np.asarray(
            [float(row["pressure_relative_l2_error"]) for row in rows],
            dtype=float,
        ),
        "lift_error": np.asarray(
            [float(row["lift_relative_l2_error"]) for row in rows],
            dtype=float,
        ),
    }


def _boxplot_panel(axis, groups, positions, colour, title):
    boxplot = axis.boxplot(
        groups,
        positions=positions,
        widths=0.65,
        patch_artist=True,
        showmeans=True,
        meanprops={
            "marker": "D",
            "markerfacecolor": "white",
            "markeredgecolor": "black",
            "markersize": 3,
        },
        medianprops={"color": "black", "linewidth": 1.3},
    )
    for box in boxplot["boxes"]:
        box.set_facecolor(colour)
        box.set_alpha(0.65)
    axis.set_ylabel("Relative L2 error")
    axis.yaxis.set_major_formatter(PercentFormatter(xmax=1.0))
    axis.set_ylim(bottom=0.0)
    axis.set_title(title)
    axis.grid(axis="y", alpha=0.3)


def save_grouped_boxplots(
    data,
    group_key,
    output_path,
    x_label,
    figure_title,
    side_by_side,
):
    group_values = np.round(data[group_key], OMEGA_DECIMALS)
    levels = np.unique(group_values)
    pressure_groups = [
        data["pressure_error"][group_values == level]
        for level in levels
    ]
    lift_groups = [
        data["lift_error"][group_values == level]
        for level in levels
    ]
    labels = [
        f"{level:+.3f}\n(n={group.size})"
        for level, group in zip(levels, pressure_groups)
    ]
    positions = np.arange(levels.size)

    if side_by_side:
        fig, axes = plt.subplots(1, 2, figsize=(14, 6), constrained_layout=True)
    else:
        figure_width = max(18.0, 0.55 * levels.size)
        fig, axes = plt.subplots(
            2,
            1,
            figsize=(figure_width, 11),
            sharex=True,
            constrained_layout=True,
        )

    _boxplot_panel(
        axes[0],
        pressure_groups,
        positions,
        "tab:blue",
        "Full-field pressure relative L2 error",
    )
    _boxplot_panel(
        axes[1],
        lift_groups,
        positions,
        "tab:green",
        "Lift-trajectory relative L2 error",
    )

    for axis in axes if side_by_side else [axes[-1]]:
        axis.set_xticks(positions)
        axis.set_xticklabels(
            labels,
            rotation=60 if levels.size > 12 else 0,
            ha="right" if levels.size > 12 else "center",
        )
        axis.set_xlabel(x_label)

    fig.suptitle(figure_title, fontsize=15)
    fig.savefig(output_path, dpi=200)
    plt.close(fig)


def main():
    parser = argparse.ArgumentParser(
        description="Regenerate SHRED boxplots from persisted candidate data."
    )
    parser.add_argument(
        "csv_path",
        nargs="?",
        type=Path,
        help="Candidate CSV; defaults to the aggregate across all training runs.",
    )
    parser.add_argument(
        "--include-warmup",
        action="store_true",
        help="Include candidates before STATISTICS_START_TIMESTEP.",
    )
    parser.add_argument(
        "--output-dir",
        type=Path,
        help="Output directory; defaults to <run>/reconstructed_plots.",
    )
    args = parser.parse_args()

    csv_path = args.csv_path or default_candidate_csv()
    output_dir = args.output_dir or csv_path.parent / "reconstructed_plots"
    output_dir.mkdir(parents=True, exist_ok=True)
    data = load_candidate_statistics(csv_path, args.include_warmup)

    delta_output = output_dir / "relative_error_by_delta_omega.png"
    save_grouped_boxplots(
        data=data,
        group_key="delta_omega",
        output_path=delta_output,
        x_label=r"Control increment $\Delta\omega$",
        figure_title="SHRED parametric prediction error by control increment",
        side_by_side=True,
    )

    omega_output = output_dir / "relative_error_by_outgoing_omega.png"
    save_grouped_boxplots(
        data=data,
        group_key="outgoing_omega",
        output_path=omega_output,
        x_label=r"Candidate outgoing omega $\omega_{out}$",
        figure_title="SHRED parametric prediction error by candidate omega",
        side_by_side=False,
    )
    print(f"Loaded {data['outgoing_omega'].size} candidate rows from {csv_path}")
    print(f"Saved {delta_output}")
    print(f"Saved {omega_output}")


if __name__ == "__main__":
    main()
