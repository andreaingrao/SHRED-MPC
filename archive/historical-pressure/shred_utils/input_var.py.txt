# Three distinct random lift setpoints, each held for 500 samples.  Leave the
# seed as None for a different sequence on every program launch; set an integer
# to reproduce a particular statistical experiment.
LIFT_TARGET_SEED = None
LIFT_TARGET_SEGMENT_LENGTH = 500
_lift_target_candidates = [round(-1 + 0.25 * i, 1) for i in range(8)]


def generate_lift_target_sequence(seed=None):
    """Create three distinct random 500-sample lift-target segments."""
    rng = random.Random(seed)
    values = rng.sample(_lift_target_candidates, 3)
    sequence = [
        target
        for target in values
        for _ in range(LIFT_TARGET_SEGMENT_LENGTH)
    ]
    return values, sequence


LIFT_TARGET_VALUES, LIFT_TARGET = generate_lift_target_sequence(LIFT_TARGET_SEED)
