def _num_from_text(s, pattern=r"(\d+)"):
    m = re.search(pattern, s)
    return int(m.group(1))

def _reshape_field(arr, nsplit, nt, npts, name):

    arr = np.asarray(arr)
    if arr.ndim == 2 and arr.shape == (nsplit*nt, npts):
        return arr.reshape(nsplit, nt, npts)
    if arr.ndim == 3 and arr.shape == (nsplit, nt, npts):
        return arr
    raise ValueError(f"{name} has unexpected shape {arr.shape} (expected {(nsplit*nt, npts)} or {(nsplit, nt, npts)})")

def _dataset_case_limit():
    if DATASET_DIMENSIONS is not None and DATASET_DIMENSIONS > 0:
        return DATASET_DIMENSIONS
    return DATASET_CASES if DATASET_CASES > 0 else None

def _select_dataset_cases(case_names):
    limit = _dataset_case_limit()
    return case_names if limit is None else case_names[:limit]

def tensor_spatial_component(tensor, dimension: int):

    tensor = tensor[..., dimension]
    return tensor

def mod_sensor_maker(states_vec, delta_t):

    zeros = np.zeros_like(states_vec[:, : delta_t,...])

    print("zeros shape")
    print(zeros.shape)
    states_vec = np.concatenate([zeros, states_vec], axis=1)

    print("states_vec shape")
    print(states_vec.shape)

    measurements_vec = []

    for i in range(states_vec.shape[0]):

        meas_t = []

        for j in range(states_vec.shape[1]):
            
            meas_delta_t = []

            for t in range(delta_t):

                meas = states_vec[i, j - t, sensor_ids, ...]
                
                meas_delta_t.append(meas)
            
            meas_t.append(meas_delta_t)
        
        measurements_vec.append(meas_t)
    
    measurements_vec = np.asarray(measurements_vec)

    return measurements_vec

def shred_decoding(manager_pod, model, decode_measurements):

    engine_pod = ParametricSHREDEngine(manager_pod, model)

    print("\n" + "-" * 28 + "\nConstructing latent space...\n" + "-" * 28 +"\n")
    latent = engine_pod.sensor_to_latent(decode_measurements) # cambiando questi valori di sensor measurements ricostruisco spazi diversi (anche non noti precedentemente)

    print("\n" + "-" * 24 + "\nDecoding latent space...\n" + "-" * 24 +"\n")
    recon  = engine_pod.decode(latent)

    print("[Decode Finished]\n")

    return recon

def data_export(states_vec, points_vec, parameters_vec, recon, sensor_ids):

        parameters_vec  = np.load(os.path.join(numpy_dir, "parameters_vec.npy"))

        def _load_npz(fname):
            with np.load(fname, allow_pickle=False) as data:
                return {k: np.array(v) for k, v in data.items()}

        recon = _load_npz(os.path.join(NUMPY_DIRECTORY, "recon.npz"))

        return (states_vec, measurement_ids, points_vec,
                parameters_vec, recon)

def energy_dissipated(state_x, state_y) -> float:
    sx = np.asarray(state_x)
    sy = np.asarray(state_y)
    if sx.shape != sy.shape:
        raise ValueError(f"shape mismatch: {sx.shape} vs {sy.shape}")
    if sx.ndim == 1:
        sx = sx[None, :]
        sy = sy[None, :]
    else:
        sx = sx.reshape(sx.shape[0], -1)
        sy = sy.reshape(sy.shape[0], -1)
    diff_x = np.diff(sx, axis=1)   # spatial diff per frame
    diff_y = np.diff(sy, axis=1)
    return float(np.sqrt(diff_x**2 + diff_y**2).sum())  # sums all timesteps

def lift_generated(state_x, state_y) -> float:
    sx = np.asarray(state_x)
    sy = np.asarray(state_y)
    if sx.shape != sy.shape:
        raise ValueError(f"shape mismatch: {sx.shape} vs {sy.shape}")
    if sx.ndim == 1:
        sx = sx[None, :]
        sy = sy[None, :]
    else:
        sx = sx.reshape(sx.shape[0], -1)
        sy = sy.reshape(sy.shape[0], -1)

    u_inf = 0.5  # OpenFoam/0/U and MPC_OpenFoam/0/U inlet velocity: uniform (0.5 0 0)
    n_points = sx.shape[1]

    cache = getattr(lift_generated, "_cache", {})
    if n_points not in cache:
        case_dir = os.path.join(DATA_DIRECTORY, "case_omega0")
        sample_file = os.path.join(case_dir, "datacsv_0.csv")

        if not os.path.isfile(sample_file):
            case_names = [
                case for case in os.listdir(DATA_DIRECTORY)
                if case.startswith("case_omega")
                and os.path.isdir(os.path.join(DATA_DIRECTORY, case))
            ]
            case_names.sort(key=lambda s: _num_from_text(s, r"case_omega(\d+)"))
            if not case_names:
                raise FileNotFoundError(f"No case_omega folders found in {DATA_DIRECTORY}")

            case_dir = os.path.join(DATA_DIRECTORY, case_names[0])
            files_list = [
                file for file in os.listdir(case_dir)
                if file.startswith("datacsv_") and file.endswith(".csv")
            ]
            files_list.sort(key=lambda s: _num_from_text(s, r"datacsv_(\d+)\.csv"))
            if not files_list:
                raise FileNotFoundError(f"No datacsv_*.csv files found in {case_dir}")
            sample_file = os.path.join(case_dir, files_list[0])

        points = pd.read_csv(sample_file)[["Points:0", "Points:1"]].to_numpy(dtype=float)
        if points.shape[0] != n_points:
            raise ValueError(
                f"point count mismatch: input has {n_points} points, "
                f"but {sample_file} has {points.shape[0]}"
            )

        center = np.asarray((0.0, 0.0), dtype=float)
        radius = np.linalg.norm(points - center, axis=1)
        sorted_ids = np.argsort(radius)
        sorted_radius = radius[sorted_ids]

        search_count = min(240, sorted_radius.size)
        radius_gap = np.diff(sorted_radius[:search_count])
        positive_gap = radius_gap[radius_gap > 1e-12]
        gap_scale = np.median(positive_gap[: min(60, positive_gap.size)]) if positive_gap.size else 0.0
        gap_threshold = max(0.05, 8.0 * gap_scale)
        ring_breaks = np.flatnonzero(radius_gap > gap_threshold)
        ring_count = int(ring_breaks[0] + 1) if ring_breaks.size else min(120, n_points)

        surface_ids = sorted_ids[:ring_count]
        theta = np.mod(np.arctan2(points[surface_ids, 1], points[surface_ids, 0]), 2.0 * np.pi)
        theta_order = np.argsort(theta)
        surface_ids = surface_ids[theta_order]
        theta = theta[theta_order]

        dtheta_prev = np.mod(theta - np.roll(theta, 1), 2.0 * np.pi)
        dtheta_next = np.mod(np.roll(theta, -1) - theta, 2.0 * np.pi)
        theta_weights = 0.5 * (dtheta_prev + dtheta_next)
        sin_theta = np.sin(theta)

        cache[n_points] = surface_ids, theta_weights, sin_theta
        lift_generated._cache = cache

    surface_ids, theta_weights, sin_theta = cache[n_points]
    speed = np.sqrt(sx[:, surface_ids]**2 + sy[:, surface_ids]**2)
    cp = 1.0 - (speed / u_inf)**2

    # Positive lift is upward; pressure acts opposite to the outward normal.
    lift_per_frame = -np.sum(cp * sin_theta[None, :] * theta_weights[None, :], axis=1)
    return float(lift_per_frame.sum())

def _pressure_lift_geometry(points):
    points = np.asarray(points, dtype=float)
    n_points = points.shape[0]
    cache = getattr(_pressure_lift_geometry, "_cache", {})

    if n_points not in cache:
        center = np.asarray((0.0, 0.0), dtype=float)
        radius = np.linalg.norm(points - center, axis=1)
        sorted_ids = np.argsort(radius)
        sorted_radius = radius[sorted_ids]

        search_count = min(240, sorted_radius.size)
        radius_gap = np.diff(sorted_radius[:search_count])
        positive_gap = radius_gap[radius_gap > 1e-12]
        gap_scale = np.median(positive_gap[: min(60, positive_gap.size)]) if positive_gap.size else 0.0
        gap_threshold = max(0.05, 8.0 * gap_scale)
        ring_breaks = np.flatnonzero(radius_gap > gap_threshold)
        ring_count = int(ring_breaks[0] + 1) if ring_breaks.size else min(120, n_points)

        surface_ids = sorted_ids[:ring_count]
        theta = np.mod(np.arctan2(points[surface_ids, 1], points[surface_ids, 0]), 2.0 * np.pi)
        theta_order = np.argsort(theta)
        surface_ids = surface_ids[theta_order]
        theta = theta[theta_order]

        dtheta_prev = np.mod(theta - np.roll(theta, 1), 2.0 * np.pi)
        dtheta_next = np.mod(np.roll(theta, -1) - theta, 2.0 * np.pi)
        theta_weights = 0.5 * (dtheta_prev + dtheta_next)
        sin_theta = np.sin(theta)

        x = points[:, 0]
        left_x = np.min(x)
        x_tol = max(1e-9, np.ptp(x) * 1e-6)
        left_ids = np.flatnonzero(x <= left_x + x_tol)
        if left_ids.size == 0:
            left_ids = np.asarray([int(np.argmin(x))])

        cache[n_points] = surface_ids, theta_weights, sin_theta, left_ids
        _pressure_lift_geometry._cache = cache

    return cache[n_points]


def lift_time_series_p(state_p, points_vec=None):
    sp = np.asarray(state_p, dtype=float)
    if sp.ndim == 1:
        sp = sp[None, :]
    elif sp.ndim > 2:
        sp = sp.reshape(-1, sp.shape[-1])

    n_points = sp.shape[1]
    points = _load_reference_points(n_points) if points_vec is None else np.asarray(points_vec, dtype=float)
    if points.shape[0] != n_points:
        raise ValueError(
            f"point count mismatch: pressure input has {n_points} points, "
            f"but points_vec has {points.shape[0]}"
        )

    surface_ids, theta_weights, sin_theta, _ = _pressure_lift_geometry(points)
    cp = sp[:, surface_ids]

    # state_p is already (p - p_inf) / q_inf. Positive lift is upward;
    # pressure acts opposite to the outward normal.
    return -np.sum(cp * sin_theta[None, :] * theta_weights[None, :], axis=1)


def lift_generated_p(state_p, points_vec=None) -> float:
    return float(lift_time_series_p(state_p, points_vec=points_vec).sum())

def _loss_savgol_smooth(loss_pred_hist, preferred_window=15, polyorder=2, median_window=5):
    """Savitzky-Golay smoothing with optional median prefilter; returns None if not applicable."""
    loss_pred_hist = np.asarray(loss_pred_hist, dtype=float)
    n = loss_pred_hist.size
    if n < polyorder + 2:
        return None

    try:
        from scipy.signal import savgol_filter, medfilt
    except ImportError:
        return None

    # optional median to knock down spikes
    if median_window and median_window > 1 and n >= median_window:
        if median_window % 2 == 0:
            median_window -= 1
        loss_pred_hist = medfilt(loss_pred_hist, kernel_size=median_window)

    # pick an odd window length that fits the data and is not too small
    window = min(preferred_window, n)
    if window % 2 == 0:
        window -= 1
    min_window = polyorder + 1
    if min_window % 2 == 0:
        min_window += 1
    if window < min_window:
        window = min_window
    if window > n:
        return None

    try:
        return savgol_filter(loss_pred_hist, window_length=window, polyorder=polyorder, mode="interp")
    except (ValueError, np.linalg.LinAlgError):
        return None

def shred_comp_plot(x, y, x_meas, y_meas, mag_x_true, mag_y_true, mag_x_pred, mag_y_pred, vmax, par_true, par_pred, timestep, type: Literal["show", "save"], reset_history: bool = False):

    # vmax = float(max(mag_true.max(), mag_pred.max()))

    mag_true = np.sqrt(mag_x_true**2 + mag_y_true**2)

    loss_pred = energy_dissipated(mag_x_pred, mag_y_pred)
    loss_true = energy_dissipated(mag_x_true, mag_y_true)

def val_time_plot(loss_vec_pred, loss_vec_cfd, type: Literal["show", "save"], out_dir="shred_images"):
    loss_vec_pred = np.asarray(loss_vec_pred, dtype=float)
    loss_vec_cfd = np.asarray(loss_vec_cfd, dtype=float)

    n = min(loss_vec_pred.shape[0], loss_vec_cfd.shape[0])
    if n == 0:
        print("val_time_plot: empty loss vectors, nothing to plot.")
        return
    if loss_vec_pred.shape[0] != loss_vec_cfd.shape[0]:
        print(f"val_time_plot: clipping to {n} points (pred={loss_vec_pred.shape[0]}, cfd={loss_vec_cfd.shape[0]}).")

    x = np.arange(n)
    diff = loss_vec_pred[:n] - loss_vec_cfd[:n]
    denom = loss_vec_cfd[:n]
    error_pct = np.where(denom != 0, diff / denom * 100.0, np.nan)
    if np.any(denom == 0):
        print("val_time_plot: zero CFD values found; percentage error set to NaN for those points.")

    fig, axs = plt.subplots(2, 1, figsize=(12, 9), sharex=True, constrained_layout=True)
    ax0, ax1 = axs

    ax0.plot(x, loss_vec_cfd[:n], label="CFD loss")
    ax0.plot(x, loss_vec_pred[:n], label="Pred loss")
    ax0.set_ylabel("Loss")
    ax0.legend()
    ax0.grid(True, alpha=0.3)

    ax1.plot(x, error_pct, label="Pred - CFD (%)", color="tab:red")
    ax1.axhline(0.0, color="gray", linewidth=0.8)
    ax1.set_xlabel("Step")
    ax1.set_ylabel("Error (%)")
    ax1.legend()
    ax1.grid(True, alpha=0.3)

    if type == "show":
        plt.show()
    elif type == "save":
        os.makedirs(out_dir, exist_ok=True)
        plt.savefig(f"{out_dir}/loss_compare.png", dpi=200)
        plt.close('all')

def _plot_desired_lift(ax, x, desired_lift, reference_steps=None):
    """Plot a scalar setpoint or a setpoint schedule at its reference steps."""
    desired_lift = np.asarray(desired_lift, dtype=float).reshape(-1)

    if desired_lift.size == 1:
        ax.axhline(
            desired_lift[0],
            label="Desired lift",
            color="gray",
            linestyle="--",
            linewidth=1.5,
        )
        return

    if reference_steps is None:
        if desired_lift.size != len(x):
            raise ValueError(
                "A desired-lift schedule must have one value per plotted point "
                "when reference_steps is not provided."
            )
        desired_lift_plot = desired_lift
    else:
        reference_steps = np.asarray(reference_steps, dtype=int)
        if reference_steps.size != len(x):
            raise ValueError("reference_steps must have one value per plotted point.")
        if reference_steps.size and (
            reference_steps.min() < 0 or reference_steps.max() >= desired_lift.size
        ):
            raise ValueError("The desired-lift schedule does not cover all reference steps.")
        desired_lift_plot = desired_lift[reference_steps]

    # 'post' holds the old objective until the next reference step, where the
    # new objective becomes active.
    ax.step(
        x,
        desired_lift_plot,
        where="post",
        label="Desired lift",
        color="gray",
        linestyle="--",
        linewidth=1.5,
    )

    if n == 0:
        print("lift_time_plot: empty lift vectors, nothing to plot.")
        return
    if lift_vec_pred.shape[0] != lift_vec_cfd.shape[0]:
        print(f"lift_time_plot: clipping to {n} points (pred={lift_vec_pred.shape[0]}, cfd={lift_vec_cfd.shape[0]}).")

        desired_lift = LIFT_TARGET

    ax0.set_title("Lift over time")
    ax0.legend()
    ax0.grid(True, alpha=0.3)

    ax1.plot(x, relative_lift_error, label="(Pred - CFD) / |CFD|", color="tab:red")
    ax1.axhline(0.0, color="gray", linewidth=0.8)
    ax1.set_xlabel("Time (s)")
    ax1.set_ylabel("Relative lift error (%)")

def state_meas_decoded(state, parameters, engine_pod):

    # Takes the state, creates from it the sensor measurements, cycles over the parameters vector and decodes all its possible evolutions

    measurements_vec = sensor_maker(states_vec = state)
    measurements_flat = measurements_vec.reshape(-1)

    N = parameters.shape[0]
    meas_tiled = np.repeat(measurements_flat[None, :], N, axis=0)  

    parameters = np.linspace(-7, 1, N)[:, None]
    sensors = np.concatenate([meas_tiled, parameters], axis=-1)

    latent = engine_pod.sensor_to_latent(sensors)

    recon = engine_pod.decode(latent)

    ux_pred = []
    uy_pred = []

    for i in range (parameters.shape[0]):

        ux = np.asarray(recon["States_x"][i, :]).reshape(-1)
        uy = np.asarray(recon["States_y"][i, :]).reshape(-1)

        ux_pred.append(ux)
        uy_pred.append(uy)

        
    ux_pred = np.asarray(ux_pred)
    uy_pred = np.asarray(uy_pred)

    state_pred = np.stack([ux_pred, uy_pred], axis= -1)

    return state_pred

def sensors_equal_distribution(measurements_flat, delta_max, N):

    # takes the measurements at the actual timestep, creates an equally spaced distribution 
    # of parameters to reconstruct from, concatenates them all for decoding

    meas_tiled = np.repeat(measurements_flat[None, :], N, axis=0)  

    parameters_distributed = np.linspace(omega0 - delta_max,  + delta_max, N)[:, None]

    sensors = np.concatenate([meas_tiled, parameters_distributed], axis=-1)

    return sensors, parameters_distributed

def decode_distributed_states(sensors, parameters_distributed, engine_pod):

    # Note that I would already have the information about parameters distributed inside sensors, 
    # however it would be more computationally expensive to extract them than what it is to just recall them here

    latent = engine_pod.sensor_to_latent(sensors)

    recon = engine_pod.decode(latent)

    ux_pred = []
    uy_pred = []

    for i in range (parameters_distributed.shape[0]):

        ux = np.asarray(recon["States_x"][i, :]).reshape(-1)
        uy = np.asarray(recon["States_y"][i, :]).reshape(-1)
        
        ux_pred.append(ux)
        uy_pred.append(uy)

        
    ux_pred = np.asarray(ux_pred)
    uy_pred = np.asarray(uy_pred)

    return ux_pred, uy_pred

import numpy as np

def get_surface_sensors(n_sensors, points_vec, R=0.5, center=(0.0, 0.0), 
                        inner_offset=0.02, outer_offset=0.05,
                        angle_range_deg=(-180, 180), random_selection=False):
    """
    Selects N sensor IDs in a ring slightly outwards from the cylinder surface.
    
    Parameters:
    - n_sensors: Desired number of sensors (int).
    - points_vec: Array of cell center coordinates, shape (N_cells, 2) or (N_cells, 3).
    - R: Radius of the cylinder.
    - center: (x, y) coordinates of the cylinder center.
    - inner_offset: Distance from the surface to skip the first/second crown of cells.
    - outer_offset: Outer bound of the acceptable sensor ring.
    - angle_range_deg: Tuple (min_angle, max_angle) in degrees to restrict placement. 
                       E.g., (-60, 60) for the wake.
    - random_selection: If True, picks sensors randomly in the region. If False, spaces them evenly.
    
    Returns:
    - List of selected sensor IDs (indices corresponding to points_vec).
    """
    x = points_vec[:, 0] - center[0]
    y = points_vec[:, 1] - center[1]
    
    # Calculate radial distance and angle for all mesh points
    radii = np.sqrt(x**2 + y**2)
    angles = np.arctan2(y, x)
    
    # Convert target angles to radians
    min_angle_rad = np.radians(angle_range_deg[0])
    max_angle_rad = np.radians(angle_range_deg[1])
    
    # Define the radial band
    min_r = R + inner_offset
    max_r = R + outer_offset
    
    # Helper to check if an angle falls within the specified range (handling wrap-around)
    def is_angle_between(target, min_a, max_a):
        target = target % (2 * np.pi)
        min_a = min_a % (2 * np.pi)
        max_a = max_a % (2 * np.pi)
        if min_a < max_a:
            return (target >= min_a) & (target <= max_a)
        else:
            return (target >= min_a) | (target <= max_a)
            
    valid_angle_mask = is_angle_between(angles, min_angle_rad, max_angle_rad)
    valid_radius_mask = (radii >= min_r) & (radii <= max_r)
    
    # Indices of all mesh cells that fall into our target physical region
    candidate_indices = np.where(valid_radius_mask & valid_angle_mask)[0]
    
    if len(candidate_indices) < n_sensors:
        raise ValueError(f"Only found {len(candidate_indices)} cells in this region. "
                         "Expand angle_range_deg, increase outer_offset, or request fewer sensors.")
                         
    # --- BRANCH 1: Random Selection ---
    if random_selection:
        # Pick N random sensors without replacement
        selected_sensor_ids = np.random.choice(candidate_indices, size=n_sensors, replace=False).tolist()
        return selected_sensor_ids

    # --- BRANCH 2: Evenly Distributed Selection ---
    candidate_angles = angles[candidate_indices]
    
    # Calculate evenly spaced targets based on the requested wedge
    if angle_range_deg[1] - angle_range_deg[0] >= 360:
        target_angles = np.linspace(min_angle_rad, min_angle_rad + 2*np.pi, n_sensors, endpoint=False)
    else:
        if n_sensors == 1:
            target_angles = [(min_angle_rad + max_angle_rad) / 2]
        else:
            diff = (max_angle_rad - min_angle_rad) % (2 * np.pi)
            target_angles = [min_angle_rad + i * (diff / (n_sensors - 1)) for i in range(n_sensors)]
    
    selected_sensor_ids = []
    
    for target in target_angles:
        # Calculate angular distance to the target, handling -pi to pi boundary
        angle_diffs = np.abs(np.arctan2(np.sin(candidate_angles - target), 
                                        np.cos(candidate_angles - target)))
        
        # Sort candidates by closest angle
        sorted_candidates_idx = np.argsort(angle_diffs)
        
        for idx in sorted_candidates_idx:
            original_idx = candidate_indices[idx]
            if original_idx not in selected_sensor_ids:
                selected_sensor_ids.append(original_idx)
                break
                
    return selected_sensor_ids

def _foam_float(value):
    text = f"{float(value):.12g}"
    return "0" if text == "-0" else text


def _foam_time_directories(case_dir):
    times = []
    for name in os.listdir(case_dir):
        path = os.path.join(case_dir, name)
        if not os.path.isdir(path):
            continue
        try:
            times.append((float(name), name))
        except ValueError:
            continue
    times.sort(key=lambda item: item[0])
    return times


def _latest_foam_time(case_dir):
    times = _foam_time_directories(case_dir)
    if not times:
        raise FileNotFoundError(f"No OpenFOAM time directories found in {case_dir}")
    return times[-1]


def _foam_field_path(case_dir, time_name, field_name):
    plain = os.path.join(case_dir, time_name, field_name)
    compressed = plain + ".gz"
    if os.path.isfile(plain):
        return plain
    if os.path.isfile(compressed):
        return compressed
    raise FileNotFoundError(f"Could not find {field_name} or {field_name}.gz in {case_dir}/{time_name}")


def _foam_field_validation_error(case_dir, time_name, field_name):
    """Return None for a readable field, otherwise a short failure reason."""
    try:
        path = _foam_field_path(case_dir, time_name, field_name)
    except FileNotFoundError:
        return f"missing {field_name}"
    if os.path.getsize(path) == 0:
        return f"empty {os.path.basename(path)}"
    try:
        opener = gzip.open if path.endswith(".gz") else open
        with opener(path, "rb") as field_file:
            while field_file.read(1024 * 1024):
                pass
    except (OSError, EOFError) as error:
        return f"unreadable {os.path.basename(path)} ({error})"
    return None


def prepare_openfoam_resume_checkpoint(
    case_dir="MPC_OpenFoam",
    required_fields=("p", "U", "phi"),
):
    """Quarantine incomplete trailing times and return the latest valid time.

    An interrupted solver can leave the numerically latest directory only
    partially written. OpenFOAM's ``latestTime`` would still select it and then
    fail while reading the missing fields. Incomplete trailing directories are
    moved under ``incomplete_time_backups`` so recovery is reversible.
    """
    case_dir = os.path.abspath(case_dir)
    times = _foam_time_directories(case_dir)
    if not times:
        raise FileNotFoundError(f"No OpenFOAM time directories found in {case_dir}")

    incomplete_trailing_times = []
    latest_complete = None
    for time_value, time_name in reversed(times):
        errors = [
            error
            for field_name in required_fields
            if (
                error := _foam_field_validation_error(
                    case_dir,
                    time_name,
                    field_name,
                )
            )
            is not None
        ]
        if not errors:
            latest_complete = (time_value, time_name)
            break
        incomplete_trailing_times.append((time_name, errors))

    if latest_complete is None:
        details = "; ".join(
            f"{time_name}: {', '.join(errors)}"
            for time_name, errors in incomplete_trailing_times
        )
        raise FileNotFoundError(
            f"No complete OpenFOAM checkpoint with {required_fields} in "
            f"{case_dir}. Checked: {details}"
        )

    if incomplete_trailing_times:
        backup_parent = os.path.join(case_dir, "incomplete_time_backups")
        os.makedirs(backup_parent, exist_ok=True)
        backup_dir = tempfile.mkdtemp(prefix="recovered_", dir=backup_parent)
        for time_name, errors in reversed(incomplete_trailing_times):
            source = os.path.join(case_dir, time_name)
            destination = os.path.join(backup_dir, time_name)
            shutil.move(source, destination)
            print(
                f"Quarantined incomplete OpenFOAM time {time_name} to "
                f"{destination}: {', '.join(errors)}"
            )

    return latest_complete


def _read_foam_field(path):
    if path.endswith(".gz"):
        with gzip.open(path, "rt") as f:
            return f.read()
    with open(path, "r") as f:
        return f.read()


def _write_foam_field(path, text):
    if path.endswith(".gz"):
        with gzip.open(path, "wt") as f:
            f.write(text)
        return
    with open(path, "w") as f:
        f.write(text)


def _omega_table_block(omega_times, omega_values):
    lines = [
        "        omega",
        "        {",
        "            type            table;",
        "            values",
        f"            {len(omega_values)}",
        "            (",
    ]
    for time, omega in zip(omega_times, omega_values):
        lines.append(f"            ({_foam_float(time)} {_foam_float(omega)})")
    lines.extend([
        "            )",
        "            ;",
        "        }",
    ])
    return lines


def _replace_obstacle_omega_block(field_text, omega_times, omega_values):
    lines = field_text.splitlines()
    obstacle_idx = None

    for i, line in enumerate(lines):
        if line.strip() == "obstacle":
            obstacle_idx = i
            break

    if obstacle_idx is None:
        raise ValueError("Could not find obstacle boundary in U field.")

    omega_idx = None
    value_idx = None
    for i in range(obstacle_idx + 1, len(lines)):
        stripped = lines[i].strip()
        keyword = stripped.split(None, 1)[0] if stripped else ""
        if keyword == "omega":
            omega_idx = i
        elif omega_idx is not None and keyword == "value":
            value_idx = i
            break
        elif omega_idx is not None and keyword == "frontAndBack":
            break

    if omega_idx is None or value_idx is None:
        raise ValueError("Could not isolate omega block inside obstacle boundary.")

    new_lines = lines[:omega_idx] + _omega_table_block(omega_times, omega_values) + lines[value_idx:]
    return "\n".join(new_lines) + "\n"


def _write_resume_omega_table(case_dir, time_name, omega_times, omega_values):
    u_path = _foam_field_path(case_dir, time_name, "U")
    u_text = _read_foam_field(u_path)
    u_text = _replace_obstacle_omega_block(u_text, omega_times, omega_values)
    _write_foam_field(u_path, u_text)


def _clear_generated_files(case_dir):
    for rel_dir, prefixes in (("VTK", ()), ("CSV_data", ("CFD_",))):
        directory = os.path.join(case_dir, rel_dir)
        if not os.path.isdir(directory):
            continue
        for name in os.listdir(directory):
            path = os.path.join(directory, name)
            if not os.path.isfile(path):
                continue
            if rel_dir == "VTK" and name.endswith(".vtk"):
                os.remove(path)
            elif rel_dir == "CSV_data" and name.endswith(".csv") and name.startswith(prefixes):
                os.remove(path)


def _OpenFoam_launcher_profile_with_csv(omegas, omega_future, csv_script="CSV_doer.py"):
    case_dir = "MPC_OpenFoam"
    omegas = np.asarray(omegas, dtype=float)
    omega_future = np.asarray(omega_future, dtype=float)

    if omegas.ndim != 1 or omegas.size == 0:
        raise ValueError(f"omegas must be a non-empty 1D sequence, got shape {omegas.shape}")
    if omega_future.ndim != 1 or omega_future.size == 0:
        raise ValueError(
            f"omega_future must be a non-empty 1D sequence, got shape {omega_future.shape}"
        )
    if not np.all(np.isfinite(omega_future)):
        raise ValueError("omega_future contains non-finite values")

    latest_time, latest_time_name = prepare_openfoam_resume_checkpoint(case_dir)

    write_dt = 1.0 / WRITE_FREQUENCY
    solver_dt = write_dt / SIM_DELTA_T_fraction
    end_time = latest_time + omega_future.size * write_dt

    current_omega = float(omegas[-1])
    omegas = np.concatenate([omegas, omega_future])

    omega_times = latest_time + np.arange(omega_future.size + 1) * write_dt
    omega_values = np.concatenate([[current_omega], omega_future])

    with open("MPC_OpenFoam/system/timesteps.txt", "w") as f:
        f.write(f"{SIM_DELTA_T_fraction}")

    with open("MPC_OpenFoam/system/total_time.txt", "w") as f:
        f.write(_foam_float(end_time))

    with open("MPC_OpenFoam/0/omega_sequence.txt", "w") as f:
        for time, omega in zip(omega_times, omega_values):
            f.write(f"({_foam_float(time)}\t{_foam_float(omega)})\n")

    _write_resume_omega_table(case_dir, latest_time_name, omega_times, omega_values)

    vtk_start_step = int(round((latest_time + write_dt) / solver_dt))
    vtk_end_step = int(round(end_time / solver_dt))
    vtk_time_range = f"{_foam_float(latest_time + write_dt)}:{_foam_float(end_time)}"

    cwd = os.getcwd()

    try:
        os.chdir(case_dir)
        _clear_generated_files(".")
        csv_env = os.environ.copy()
        csv_env["MPC_VTK_START_STEP"] = str(vtk_start_step)
        csv_env["MPC_VTK_END_STEP"] = str(vtk_end_step)
        subprocess.run(["pimpleFoam"], check=True)
        subprocess.run(["foamToVTK", "-time", vtk_time_range], check=True)
        subprocess.run(["pvpython", csv_script], check=True, env=csv_env)
    finally:
        os.chdir(cwd)

    return omegas


def _OpenFoam_launcher_with_csv(omegas, par_optimum, control_horizon=CONTROL_HORIZON, csv_script="CSV_doer.py"):
    if control_horizon <= 0:
        raise ValueError(f"control_horizon must be positive, got {control_horizon}")

    omegas = np.asarray(omegas, dtype=float)
    if omegas.ndim != 1 or omegas.size == 0:
        raise ValueError(f"omegas must be a non-empty 1D sequence, got shape {omegas.shape}")

    omega_line = np.linspace(
        float(omegas[-1]),
        float(par_optimum),
        control_horizon + 1,
    )[1:]
    return _OpenFoam_launcher_profile_with_csv(
        omegas=omegas,
        omega_future=omega_line,
        csv_script=csv_script,
    )


def OpenFoam_launcher(omegas, par_optimum, control_horizon=CONTROL_HORIZON):
    return _OpenFoam_launcher_with_csv(
        omegas=omegas,
        par_optimum=par_optimum,
        control_horizon=control_horizon,
        csv_script="CSV_doer.py",
    )


def OpenFoam_launcher_p(omegas, par_optimum, control_horizon=CONTROL_HORIZON):
    return _OpenFoam_launcher_with_csv(
        omegas=omegas,
        par_optimum=par_optimum,
        control_horizon=control_horizon,
        csv_script="CSV_doer_p.py",

def OpenFoam_launcher_profile_p(omegas, omega_future):

def OpenFoam_launcher_stat(
    omegas,
    par_optimum,
    control_horizon=CONTROL_HORIZON,
    simulation_id=0,
    cycle_id=0,
    candidate_id=0,
    output_root="MPC_OpenFoam/CSV_data_stat",
    omega_future=None,
):
    """Evaluate one pressure candidate without advancing the online CFD case.

    The current latest OpenFOAM time is snapshotted, the candidate is simulated,
    and its CSV files are copied to a cycle/candidate-specific directory.  All
    generated time directories and launcher side effects are then restored so
    the next statistical candidate starts from exactly the same checkpoint.

    Returns
    -------
    str
        Directory containing the candidate CFD fields.  When ``omega_future``
        is supplied, one CSV is produced for each entry of that profile.
    """
    if control_horizon <= 0:
        raise ValueError(f"control_horizon must be positive, got {control_horizon}")

    case_dir = os.path.abspath("MPC_OpenFoam")
    latest_time, latest_time_name = prepare_openfoam_resume_checkpoint(case_dir)
    latest_dir = os.path.join(case_dir, latest_time_name)
    initial_time_names = {name for _, name in _foam_time_directories(case_dir)}

    omegas = np.asarray(omegas, dtype=float)
    if omegas.ndim != 1 or omegas.size == 0:
        raise ValueError(f"omegas must be a non-empty 1D sequence, got shape {omegas.shape}")

    if omega_future is None:
        evaluation_horizon = int(control_horizon)
    else:
        omega_future = np.asarray(omega_future, dtype=float)
        if omega_future.ndim != 1 or omega_future.size == 0:
            raise ValueError(
                f"omega_future must be a non-empty 1D sequence, got shape {omega_future.shape}"
            )
        evaluation_horizon = int(omega_future.size)

    output_dir = os.path.abspath(
        os.path.join(
            output_root,
            f"simulation_{int(simulation_id):03d}",
            f"cycle_{int(cycle_id):04d}",
            f"candidate_{int(candidate_id):03d}_omega_{float(par_optimum):+.6f}",
        )
    )

    auxiliary_files = [
        os.path.join(case_dir, "system", "timesteps.txt"),
        os.path.join(case_dir, "system", "total_time.txt"),
        os.path.join(case_dir, "0", "omega_sequence.txt"),
    ]
    generated_directories = [
        os.path.join(case_dir, "VTK"),
        os.path.join(case_dir, "CSV_data"),
    ]

    with tempfile.TemporaryDirectory(prefix="shred_openfoam_stat_") as temp_dir:
        latest_snapshot = os.path.join(temp_dir, "latest_time")
        shutil.copytree(latest_dir, latest_snapshot)

        auxiliary_snapshots = {}
        for index, path in enumerate(auxiliary_files):
            if os.path.isfile(path):
                snapshot = os.path.join(temp_dir, f"auxiliary_{index}")
                shutil.copy2(path, snapshot)
                auxiliary_snapshots[path] = snapshot
            else:
                auxiliary_snapshots[path] = None

        generated_snapshots = {}
        for index, path in enumerate(generated_directories):
            if os.path.isdir(path):
                snapshot = os.path.join(temp_dir, f"generated_{index}")
                shutil.copytree(path, snapshot)
                generated_snapshots[path] = snapshot
            else:
                generated_snapshots[path] = None

        try:
            if omega_future is None:
                _OpenFoam_launcher_with_csv(
                    omegas=omegas,
                    par_optimum=float(par_optimum),
                    control_horizon=control_horizon,
                    csv_script="CSV_doer_p.py",
                )
            else:
                _OpenFoam_launcher_profile_with_csv(
                    omegas=omegas,
                    omega_future=omega_future,
                    csv_script="CSV_doer_p.py",
                )

            csv_source = os.path.join(case_dir, "CSV_data")
            expected_files = [
                os.path.join(csv_source, f"CFD_{i}.csv")
                for i in range(evaluation_horizon)
            ]
            missing = [path for path in expected_files if not os.path.isfile(path)]
            if missing:
                raise FileNotFoundError(
                    f"Statistical CFD candidate produced incomplete CSV output: {missing}"
                )

            if os.path.isdir(output_dir):
                shutil.rmtree(output_dir)
            os.makedirs(output_dir, exist_ok=False)
            for source in expected_files:
                shutil.copy2(source, os.path.join(output_dir, os.path.basename(source)))

            with open(os.path.join(output_dir, "metadata.json"), "w") as file:
                json.dump(
                    {
                        "cycle_id": int(cycle_id),
                        "simulation_id": int(simulation_id),
                        "candidate_id": int(candidate_id),
                        "base_cfd_time": float(latest_time),
                        "current_omega": float(omegas[-1]),
                        "terminal_omega": float(par_optimum),
                        "control_horizon": int(control_horizon),
                        "evaluation_horizon": evaluation_horizon,
                    },
                    file,
                    indent=2,
                    sort_keys=True,
                )
        finally:
            for _, time_name in _foam_time_directories(case_dir):
                if time_name not in initial_time_names:
                    shutil.rmtree(os.path.join(case_dir, time_name))

            if os.path.isdir(latest_dir):
                shutil.rmtree(latest_dir)
            shutil.copytree(latest_snapshot, latest_dir)

            for path, snapshot in auxiliary_snapshots.items():
                if os.path.exists(path):
                    os.remove(path)
                if snapshot is not None:
                    os.makedirs(os.path.dirname(path), exist_ok=True)
                    shutil.copy2(snapshot, path)

            for path, snapshot in generated_snapshots.items():
                if os.path.isdir(path):
                    shutil.rmtree(path)
                if snapshot is not None:
                    shutil.copytree(snapshot, path)

    return output_dir

def DataLoad_Training(case_key: str):

    #### DATA LOAD ####

    states_vec, points_vec, parameters_time_tensor      = data_load(case_key=case_key) 
    # states_vec: [cases, timesteps, cells, dimensions]
    # points_vec: [cells, dimensions]
    # parameters_time_tensor: [cases, timesteps, parameter]

    measurements_vec_training                           = sensor_maker(states_vec = states_vec) 
    # [cases, timesteps, sensor_ids, dimensions]


    #### TRAINING ####

    manager_pod, model, parameters_time_tensor          = shred_training(states_vec=states_vec, measurements_vec=measurements_vec_training, parameters_time_tensor=parameters_time_tensor)
    # Il parameters_time_tensor che esce è quello traslato del deltaT per la previsione

    
    #### EXPORT ####

    engine_pod                                          = ParametricSHREDEngine(manager_pod, model)
    state_0 = states_vec[0, ...]

    return engine_pod, state_0, points_vec, manager_pod
