def _candidate_row_key(row):
    return (
        row["run_id"],
        row["simulation_id"],
        row["simulation_cycle"],
        row["candidate_id"],
    )


def _validated_candidate_reader(csv_file, csv_path):
    reader = csv.DictReader(csv_file)
    if reader.fieldnames != STAT_CANDIDATE_FIELDS:
        raise ValueError(
            f"Candidate CSV schema mismatch in {csv_path}: "
            f"expected {STAT_CANDIDATE_FIELDS}, got {reader.fieldnames}"
        )
    return reader


def reconcile_aggregate_statistics():
    """Import rows missing from the master CSV without duplicating cycles."""
    aggregate_path = Path(STAT_AGGREGATE_CSV)
    if not aggregate_path.exists():
        with aggregate_path.open("x", newline="") as csv_file:
            writer = csv.DictWriter(csv_file, fieldnames=STAT_CANDIDATE_FIELDS)
            writer.writeheader()

    with aggregate_path.open(newline="") as csv_file:
        aggregate_reader = _validated_candidate_reader(csv_file, aggregate_path)
        known_keys = {_candidate_row_key(row) for row in aggregate_reader}

    missing_rows = []
    for run_csv in sorted(Path(STAT_OUTPUT_ROOT).glob("run_*/candidate_statistics.csv")):
        with run_csv.open(newline="") as csv_file:
            run_reader = _validated_candidate_reader(csv_file, run_csv)
            for row in run_reader:
                key = _candidate_row_key(row)
                if key not in known_keys:
                    missing_rows.append(row)
                    known_keys.add(key)

    if missing_rows:
        with aggregate_path.open("a", newline="") as csv_file:
            writer = csv.DictWriter(csv_file, fieldnames=STAT_CANDIDATE_FIELDS)
            writer.writerows(missing_rows)
    print(
        f"Aggregate candidate data: {len(known_keys)} rows "
        f"({len(missing_rows)} recovered from per-run CSVs)"
    )


def load_previous_runs_for_aggregate_plots():
    """Seed heatmap/boxplot arrays from all already persisted eligible rows."""
    with open(STAT_AGGREGATE_CSV, newline="") as csv_file:
        reader = _validated_candidate_reader(csv_file, STAT_AGGREGATE_CSV)
        for row in reader:
            if int(row["eligible_for_statistics"]) != 1:
                continue
            incoming_omega = float(row["incoming_omega"])
            outgoing_omega = float(row["outgoing_omega"])
            heatmap_incoming_omegas.append(incoming_omega)
            heatmap_outgoing_omegas.append(outgoing_omega)
            heatmap_state_errors.append(float(row["pressure_relative_l2_error"]))
            heatmap_lift_errors.append(float(row["lift_relative_l2_error"]))
            if int(row["shred_selected"]) == 1:
                heatmap_selected_incoming_omegas.append(incoming_omega)
                heatmap_selected_outgoing_omegas.append(outgoing_omega)
                if int(row["choice_mistake"]) == 1:
                    heatmap_wrong_incoming_omegas.append(incoming_omega)
                    heatmap_wrong_outgoing_omegas.append(outgoing_omega)

class InitialOnlineCaseSnapshot:
    """Restore the online OpenFOAM case before each independent simulation."""

    def __init__(self):
        self.case_dir = Path("MPC_OpenFoam").resolve()
        prepare_openfoam_resume_checkpoint(str(self.case_dir))
        time_directories = _foam_time_directories(str(self.case_dir))
        if not time_directories:
            raise FileNotFoundError(f"No OpenFOAM time directories in {self.case_dir}")

        self.initial_time_names = {name for _, name in time_directories}
        self.latest_time, self.latest_time_name = time_directories[-1]
        self.latest_dir = self.case_dir / self.latest_time_name
        self.temp = tempfile.TemporaryDirectory(prefix="shred_stat_initial_case_")
        self.temp_dir = Path(self.temp.name)
        self.latest_snapshot = self.temp_dir / "latest_time"
        shutil.copytree(self.latest_dir, self.latest_snapshot)

        self.auxiliary_files = [
            self.case_dir / "system" / "timesteps.txt",
            self.case_dir / "system" / "total_time.txt",
            self.case_dir / "0" / "omega_sequence.txt",
        ]
        self.auxiliary_snapshots = {}
        for index, path in enumerate(self.auxiliary_files):
            if path.is_file():
                snapshot = self.temp_dir / f"auxiliary_{index}"
                shutil.copy2(path, snapshot)
                self.auxiliary_snapshots[path] = snapshot
            else:
                self.auxiliary_snapshots[path] = None

        self.generated_directories = [
            self.case_dir / "VTK",
            self.case_dir / "CSV_data",
        ]
        self.generated_snapshots = {}
        for index, path in enumerate(self.generated_directories):
            if path.is_dir():
                snapshot = self.temp_dir / f"generated_{index}"
                shutil.copytree(path, snapshot)
                self.generated_snapshots[path] = snapshot
            else:
                self.generated_snapshots[path] = None

    def restore(self):
        for _, time_name in _foam_time_directories(str(self.case_dir)):
            if time_name not in self.initial_time_names:
                shutil.rmtree(self.case_dir / time_name)

        if self.latest_dir.is_dir():
            shutil.rmtree(self.latest_dir)
        shutil.copytree(self.latest_snapshot, self.latest_dir)

        for path, snapshot in self.auxiliary_snapshots.items():
            if path.exists():
                path.unlink()
            if snapshot is not None:
                path.parent.mkdir(parents=True, exist_ok=True)
                shutil.copy2(snapshot, path)

        for path, snapshot in self.generated_snapshots.items():
            if path.is_dir():
                shutil.rmtree(path)
            if snapshot is not None:
                shutil.copytree(snapshot, path)

    def close(self):
        self.temp.cleanup()


def statistical_candidate_omegas(current_omega):
    """Return the configured number of unique candidates, shifted at bounds."""
    lower = current_omega - MPC_OMEGA_SEARCH_RADIUS
    upper = current_omega + MPC_OMEGA_SEARCH_RADIUS
    full_width = 2.0 * MPC_OMEGA_SEARCH_RADIUS

    if lower < MPC_OMEGA_MIN:
        lower = MPC_OMEGA_MIN
        upper = min(MPC_OMEGA_MAX, lower + full_width)
    if upper > MPC_OMEGA_MAX:
        upper = MPC_OMEGA_MAX
        lower = max(MPC_OMEGA_MIN, upper - full_width)

    candidates = np.round(
        np.linspace(lower, upper, STAT_OMEGA_GRID_SIZE),
        OMEGA_TABLE_DECIMALS,
    )
    candidates = np.unique(candidates)
    if candidates.size != STAT_OMEGA_GRID_SIZE:
        raise ValueError(
            f"Expected {STAT_OMEGA_GRID_SIZE} unique statistical candidates, "
            f"got {candidates}"
        )
    return candidates


def relative_l2_error(prediction, reference, epsilon=STAT_RELATIVE_L2_EPSILON):
    """Return ||prediction - reference||_2 / ||reference||_2."""
    prediction = np.asarray(prediction, dtype=float)
    reference = np.asarray(reference, dtype=float)
    if prediction.shape != reference.shape:
        raise ValueError(
            "Relative L2 inputs must have identical shapes, got "
            f"{prediction.shape} and {reference.shape}"
        )

    reference_norm = float(np.linalg.norm(reference.ravel()))
    error_norm = float(np.linalg.norm((prediction - reference).ravel()))
    return error_norm / max(reference_norm, epsilon)


def save_statistical_evolution_plot():
    cycles = np.asarray(stat_cycle_vec)
    state_median = np.asarray(state_error_median_history)
    state_q1 = np.asarray(state_error_q1_history)
    state_q3 = np.asarray(state_error_q3_history)
    lift_median = np.asarray(lift_error_median_history)
    lift_q1 = np.asarray(lift_error_q1_history)
    lift_q3 = np.asarray(lift_error_q3_history)
    mistakes = np.asarray(parameter_mistake_history)
    simulation_ids = np.asarray(stat_simulation_vec)

    fig, axes = plt.subplots(3, 1, figsize=(10, 12), sharex=True, constrained_layout=True)
    fig.suptitle(f"Current trained model: {active_training_id}", fontsize=14)

    def plot_distribution_summary(
        axis,
        median,
        q1,
        q3,
        colour,
        title,
    ):
        axis.fill_between(
            cycles,
            q1,
            q3,
            color=colour,
            alpha=0.20,
            label="Cumulative IQR (25th-75th percentile)",
        )
        axis.plot(
            cycles,
            median,
            color=colour,
            linewidth=2.0,
            label="Cumulative median",
        )
        axis.text(
            0.99,
            0.03,
            f"Final median: {median[-1]:.2%}   |   "
            f"IQR: {q1[-1]:.2%}-{q3[-1]:.2%}",
            transform=axis.transAxes,
            ha="right",
            va="bottom",
            fontsize=9,
            bbox={"facecolor": "white", "alpha": 0.8, "edgecolor": "0.8"},
        )
        axis.set_ylabel("Relative L2 error")
        axis.yaxis.set_major_formatter(PercentFormatter(xmax=1.0))
        axis.set_ylim(bottom=0.0)
        axis.set_title(title)
        axis.grid(True, alpha=0.3)
        axis.legend(loc="best")

    plot_distribution_summary(
        axes[0],
        state_median,
        state_q1,
        state_q3,
        "tab:blue",
        "Cumulative full-field pressure-error distribution",
    )
    plot_distribution_summary(
        axes[1],
        lift_median,
        lift_q1,
        lift_q3,
        "tab:green",
        "Cumulative lift-trajectory error distribution",
    )

    cumulative_mistakes = np.cumsum(mistakes)
    axes[2].step(
        cycles,
        cumulative_mistakes,
        where="post",
        color="tab:purple",
        label="Cumulative wrong parameter choices",
    )
    wrong_cycles = cycles[mistakes.astype(bool)]
    if wrong_cycles.size:
        axes[2].scatter(
            wrong_cycles,
            cumulative_mistakes[mistakes.astype(bool)],
            color="tab:red",
            marker="x",
            s=70,
            label="Wrong choice this cycle",
        )
    axes[2].set_xlabel("MPC cycle")
    axes[2].set_ylabel("Cumulative count")
    axes[2].set_title("SHRED choice versus CFD-optimal choice")
    axes[2].grid(True, alpha=0.3)
    axes[2].legend(loc="best")

    if simulation_ids.size > 1:
        simulation_changes = np.flatnonzero(np.diff(simulation_ids) != 0) + 1
        for change_index in simulation_changes:
            boundary = 0.5 * (cycles[change_index - 1] + cycles[change_index])
            for axis in axes:
                axis.axvline(boundary, color="black", linestyle=":", alpha=0.5)

    fig.savefig(f"{STAT_OUTPUT_DIR}/predictor_statistical_evolution.png", dpi=200)
    plt.close(fig)


def save_delta_omega_error_boxplots():
    """Plot candidate relative errors grouped by the applied omega change."""
    incoming = np.asarray(heatmap_incoming_omegas, dtype=float)
    outgoing = np.asarray(heatmap_outgoing_omegas, dtype=float)
    pressure_errors = np.asarray(heatmap_state_errors, dtype=float)
    lift_errors = np.asarray(heatmap_lift_errors, dtype=float)
    if not (
        incoming.size
        == outgoing.size
        == pressure_errors.size
        == lift_errors.size
    ):
        raise ValueError("Delta-omega boxplot arrays must have equal lengths")
    if incoming.size == 0:
        return

    delta_omegas = np.round(outgoing - incoming, OMEGA_TABLE_DECIMALS)
    delta_levels = np.unique(delta_omegas)
    pressure_groups = [pressure_errors[delta_omegas == value] for value in delta_levels]
    lift_groups = [lift_errors[delta_omegas == value] for value in delta_levels]
    positions = np.arange(delta_levels.size)
    labels = [
        f"{value:+.3f}\n(n={group.size})"
        for value, group in zip(delta_levels, pressure_groups)
    ]

    fig, axes = plt.subplots(1, 2, figsize=(14, 6), constrained_layout=True)
    plot_settings = (
        (axes[0], pressure_groups, "Full-field pressure relative L2 error", "tab:blue"),
        (axes[1], lift_groups, "Lift-trajectory relative L2 error", "tab:green"),
    )
    for axis, groups, title, colour in plot_settings:
        boxplot = axis.boxplot(
            groups,
            positions=positions,
            widths=0.65,
            patch_artist=True,
            showmeans=True,
            meanprops={
                "marker": "D",
                "markerfacecolor": "white",
                "markeredgecolor": "black",
                "markersize": 4,
            },
            medianprops={"color": "black", "linewidth": 1.5},
        )
        for box in boxplot["boxes"]:

def binned_omega_error_statistics(
    incoming_omegas,
    outgoing_omegas,
    state_errors,
    lift_errors,
    wrong_incoming_omegas,
    wrong_outgoing_omegas,
    selected_incoming_omegas,
    selected_outgoing_omegas,
):
    """Aggregate candidate errors and choice failures by omega transition."""
    incoming_omegas = np.asarray(incoming_omegas, dtype=float)
    outgoing_omegas = np.asarray(outgoing_omegas, dtype=float)
    state_errors = np.asarray(state_errors, dtype=float)
    lift_errors = np.asarray(lift_errors, dtype=float)

    if not (
        incoming_omegas.size
        == outgoing_omegas.size
        == state_errors.size
        == lift_errors.size
    ):
        raise ValueError("Heatmap candidate arrays must have equal lengths")

    omega_span = MPC_OMEGA_MAX - MPC_OMEGA_MIN
    number_of_intervals = int(round(omega_span / STAT_OMEGA_BIN_WIDTH))
    if not np.isclose(
        number_of_intervals * STAT_OMEGA_BIN_WIDTH,
        omega_span,
        rtol=0.0,
        atol=10 ** (-OMEGA_TABLE_DECIMALS),
    ):
        raise ValueError(
            "The omega range must be divisible by the statistical candidate "
            f"spacing ({STAT_OMEGA_BIN_WIDTH})"
        )

    # One equal-width heatmap cell per attainable statistical omega level.
    # Half-step outer edges ensure that omega == 0 has its own cell rather
    # than sharing a coarse interval with neighbouring values.
    centers = np.linspace(
        MPC_OMEGA_MIN,
        MPC_OMEGA_MAX,
        number_of_intervals + 1,
    )
    edges = np.concatenate(
        (
            [centers[0] - 0.5 * STAT_OMEGA_BIN_WIDTH],
            centers + 0.5 * STAT_OMEGA_BIN_WIDTH,
        )
    )
    number_of_bins = centers.size

    def bin_ids(values):
        ids = np.searchsorted(edges, values, side="right") - 1
        return np.clip(ids, 0, number_of_bins - 1)

    incoming_ids = bin_ids(incoming_omegas)
    outgoing_ids = bin_ids(outgoing_omegas)

    shape = (number_of_bins, number_of_bins)
    state_median = np.full(shape, np.nan)
    lift_median = np.full(shape, np.nan)
    state_std = np.full(shape, np.nan)
    lift_std = np.full(shape, np.nan)
    sample_count = np.zeros(shape, dtype=int)

    # Matrix indexing is [outgoing bin, incoming bin] so the plotted vertical
    # axis is outgoing omega and the horizontal axis is incoming omega.
    for outgoing_id in range(number_of_bins):
        for incoming_id in range(number_of_bins):
            mask = (incoming_ids == incoming_id) & (outgoing_ids == outgoing_id)
            if not np.any(mask):
                continue
            state_values = state_errors[mask]
            lift_values = lift_errors[mask]
            state_median[outgoing_id, incoming_id] = float(np.median(state_values))
            lift_median[outgoing_id, incoming_id] = float(np.median(lift_values))
            state_std[outgoing_id, incoming_id] = float(np.std(state_values))
            lift_std[outgoing_id, incoming_id] = float(np.std(lift_values))
            sample_count[outgoing_id, incoming_id] = int(np.count_nonzero(mask))

    selected_incoming_omegas = np.asarray(selected_incoming_omegas, dtype=float)
    selected_outgoing_omegas = np.asarray(selected_outgoing_omegas, dtype=float)
    if selected_incoming_omegas.size != selected_outgoing_omegas.size:
        raise ValueError("Selected-choice omega arrays must have equal lengths")

    selection_count = np.zeros(shape, dtype=int)
    if selected_incoming_omegas.size:
        selected_incoming_ids = bin_ids(selected_incoming_omegas)
        selected_outgoing_ids = bin_ids(selected_outgoing_omegas)
        np.add.at(
            selection_count,
            (selected_outgoing_ids, selected_incoming_ids),
            1,
        )

    wrong_count = np.zeros(shape, dtype=int)
    wrong_incoming_omegas = np.asarray(wrong_incoming_omegas, dtype=float)
    wrong_outgoing_omegas = np.asarray(wrong_outgoing_omegas, dtype=float)
    if wrong_incoming_omegas.size != wrong_outgoing_omegas.size:
        raise ValueError("Wrong-choice omega arrays must have equal lengths")
    if wrong_incoming_omegas.size:
        wrong_incoming_ids = bin_ids(wrong_incoming_omegas)
        wrong_outgoing_ids = bin_ids(wrong_outgoing_omegas)
        np.add.at(wrong_count, (wrong_outgoing_ids, wrong_incoming_ids), 1)

    # Failure rate is conditioned on SHRED actually selecting the transition.
    # Unselected transitions remain blank instead of being shown as zero-rate.
    visited_by_selector = selection_count > 0
    failure_rate = np.full(shape, np.nan)
    failure_rate[visited_by_selector] = (
        wrong_count[visited_by_selector] / selection_count[visited_by_selector]
    )
    wrong_count_display = np.where(visited_by_selector, wrong_count, np.nan)
    selection_count_display = np.where(
        visited_by_selector,
        selection_count,
        np.nan,
    )
    return {
        "edges": edges,
        "centers": centers,
        "state_median": state_median,
        "lift_median": lift_median,
        "state_std": state_std,
        "lift_std": lift_std,
        "failure_rate": failure_rate,
        "wrong_count": wrong_count_display,
        "selection_count": selection_count_display,
        "sample_count": sample_count,
    }

                "cfd_cost_rank": int(cfd_cost_ranks[candidate_id]),
                "shred_selected": int(candidate_id == id_omega_choice),
                "cfd_selected": int(candidate_id == id_omega_choice_cfd),
                "choice_mistake": parameter_mistake,
                "cfd_best_cost_margin": cfd_best_cost_margin,
                "selected_cfd_regret": selected_cfd_regret,
                "selected_cfd_normalized_regret": selected_cfd_normalized_regret,
            }
        )
    append_candidate_statistics(candidate_records)

    if t >= STATISTICS_START_TIMESTEP:
        cumulative_parameter_mistakes += parameter_mistake

        # Pool every eligible candidate error from every completed cycle and
        # every simulation. At the kth included cycle these contain
        # STAT_OMEGA_GRID_SIZE*k samples. Warm-up cycles
        # before STATISTICS_START_TIMESTEP never enter these arrays.
        all_state_error_candidates.extend(state_error_candidates.tolist())
        all_lift_error_candidates.extend(lift_error_candidates.tolist())
        heatmap_incoming_omegas.extend(
            [current_omega] * len(possible_omegas)
        )
        heatmap_outgoing_omegas.extend(possible_omegas.astype(float).tolist())
        heatmap_state_errors.extend(state_error_candidates.tolist())
        heatmap_lift_errors.extend(lift_error_candidates.tolist())
        heatmap_selected_incoming_omegas.append(current_omega)
        heatmap_selected_outgoing_omegas.append(
            float(possible_omegas[id_omega_choice])
        )
        if parameter_mistake:
            heatmap_wrong_incoming_omegas.append(current_omega)
            heatmap_wrong_outgoing_omegas.append(
                float(possible_omegas[id_omega_choice])
            )
        pooled_state_errors = np.asarray(all_state_error_candidates, dtype=float)
        pooled_lift_errors = np.asarray(all_lift_error_candidates, dtype=float)

        stat_cycle_vec.append(len(stat_cycle_vec) + 1)
        stat_simulation_vec.append(simulation_id)
        state_error_median_history.append(float(np.median(pooled_state_errors)))
        state_error_q1_history.append(float(np.quantile(pooled_state_errors, 0.25)))
        state_error_q3_history.append(float(np.quantile(pooled_state_errors, 0.75)))
        lift_error_median_history.append(float(np.median(pooled_lift_errors)))
        lift_error_q1_history.append(float(np.quantile(pooled_lift_errors, 0.25)))
        lift_error_q3_history.append(float(np.quantile(pooled_lift_errors, 0.75)))
        parameter_mistake_history.append(parameter_mistake)

        print(
            f"SHRED omega choice = {possible_omegas[id_omega_choice]:.6f}; "
            f"CFD omega choice = {possible_omegas[id_omega_choice_cfd]:.6f}; "
            f"counted wrong choices = {cumulative_parameter_mistakes}"
        )
        save_statistical_evolution_plot()
        save_delta_omega_error_boxplots()
        save_outgoing_omega_error_boxplots()
        save_omega_error_heatmaps()
    else:
        print(
            f"Warm-up timestep {t}: SHRED omega choice = "
            f"{possible_omegas[id_omega_choice]:.6f}; CFD omega choice = "
            f"{possible_omegas[id_omega_choice_cfd]:.6f}. Not counted."
        )

    # Compare like with like later: CFD advances only CONTROL_HORIZON samples.
    lift_pred.append(lift_applied_mean_vec[id_omega_choice])

    parameters_vec.append(float(possible_omegas[id_omega_choice]))


    #### COMPARATIVE PLOT CREATION ####

    parameters_cfd_sequence = OpenFoam_launcher_p(parameters_cfd_sequence, parameters_vec[-1], control_horizon=CONTROL_HORIZON)
    os.chdir(cwd)

    p_cfd, states_vec_cfd = pressure_cfd_extractor(count=CONTROL_HORIZON, preprocess=True)
    p_cfd_new = p_cfd
    parameter_cfd_new = parameters_cfd_sequence[-CONTROL_HORIZON:]
    timestep_new = len(parameters_cfd_sequence) - CONTROL_HORIZON

    states_vec = np.concatenate([states_vec, states_vec_cfd], axis=0)

    lift_cfd = lift_generated_p(p_cfd_new, points_vec=points_vec) / CONTROL_HORIZON

    lift_real.append(lift_cfd)

    t = timestep_new + CONTROL_HORIZON
    simulation_cycle += 1
    completed_mpc_cycles += 1

    if simulation_cycle >= STAT_N_CYCLES:
        simulation_id += 1
        if simulation_id < STAT_NUM_SIMULATIONS:
            print(
                f"Resetting OpenFOAM and Python histories for simulation "
                f"{simulation_id + 1}/{STAT_NUM_SIMULATIONS}"
            )
            initial_online_case.restore()
            states_vec = initial_states_vec.copy()
            measurements = []
            parameters_vec = [omega0]
            parameters_cfd_sequence = [omega0] * shred_lags
            lift_pred = []
            lift_real = []
            t = shred_lags
            simulation_cycle = 0
            simulation_target_values, simulation_lift_target = (
                create_simulation_target(simulation_id)
            )

    if (
        simulation_id < STAT_NUM_SIMULATIONS
        and completed_mpc_cycles % STAT_RETRAIN_EVERY_CYCLES == 0
    ):
        retrain_shred_predictor()

initial_online_case.close()
