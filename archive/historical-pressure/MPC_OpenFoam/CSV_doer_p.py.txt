# MPC_OpenFoam/CSV_doer_p.py
# Pressure variant of CSV_doer.py for pressure-based MPC feedback.

from pathlib import Path
import os

from paraview.simple import *
paraview.simple._DisableFirstRenderCameraReset()


base_dir = Path(__file__).resolve().parent
vtk_dir = base_dir / "VTK"

if not vtk_dir.is_dir():
    raise FileNotFoundError(f"VTK directory not found: {vtk_dir}")


def _timestep_key(path: Path) -> float:
    try:
        return float(path.stem.rsplit("_", 1)[1])
    except (IndexError, ValueError):
        return float("inf")


vtk_files = sorted(vtk_dir.glob("*.vtk"), key=_timestep_key)
vtk_start_step = os.environ.get("MPC_VTK_START_STEP")
vtk_end_step = os.environ.get("MPC_VTK_END_STEP")

if vtk_start_step is not None and vtk_end_step is not None:
    vtk_start_step = float(vtk_start_step)
    vtk_end_step = float(vtk_end_step)
    vtk_files = [
        path for path in vtk_files
        if vtk_start_step <= _timestep_key(path) <= vtk_end_step
    ]

if not vtk_files:
    raise FileNotFoundError(f"No VTK files found inside {vtk_dir}")

file_names = [str(path) for path in vtk_files]
registration_name = f"{vtk_files[0].stem}.vtk*"

case_vtk = LegacyVTKReader(
    registrationName=registration_name,
    FileNames=file_names,
)

animationScene1 = GetAnimationScene()
timeKeeper1 = GetTimeKeeper()
animationScene1.UpdateAnimationUsingDataTimeSteps()

renderView1 = GetActiveViewOrCreate('RenderView')
case_vtk_display = Show(case_vtk, renderView1, 'UnstructuredGridRepresentation')
case_vtk_display.Representation = 'Surface'
case_vtk_display.ColorArrayName = [None, '']
case_vtk_display.SelectTCoordArray = 'None'
case_vtk_display.SelectNormalArray = 'None'
case_vtk_display.SelectTangentArray = 'None'
case_vtk_display.OSPRayScaleArray = 'p'
case_vtk_display.OSPRayScaleFunction = 'PiecewiseFunction'
case_vtk_display.SelectOrientationVectors = 'None'
case_vtk_display.ScaleFactor = 10.0
case_vtk_display.SelectScaleArray = 'None'
case_vtk_display.GlyphType = 'Arrow'
case_vtk_display.GlyphTableIndexArray = 'None'
case_vtk_display.GaussianRadius = 0.5
case_vtk_display.SetScaleArray = ['POINTS', 'p']
case_vtk_display.ScaleTransferFunction = 'PiecewiseFunction'
case_vtk_display.OpacityArray = ['POINTS', 'p']
case_vtk_display.OpacityTransferFunction = 'PiecewiseFunction'
case_vtk_display.DataAxesGrid = 'GridAxesRepresentation'
case_vtk_display.PolarAxes = 'PolarAxesRepresentation'
case_vtk_display.ScalarOpacityUnitDistance = 4.613844601809179
case_vtk_display.OpacityArrayName = ['POINTS', 'p']
case_vtk_display.SelectInputVectors = ['POINTS', '']
case_vtk_display.WriteLog = ''

renderView1.ResetCamera(False)
renderView1.Update()

cellCenters1 = CellCenters(registrationName='CellCenters1', Input=case_vtk)
cellCenters1_display = Show(cellCenters1, renderView1, 'GeometryRepresentation')
cellCenters1_display.Representation = 'Surface'
cellCenters1_display.ColorArrayName = [None, '']
cellCenters1_display.SelectTCoordArray = 'None'
cellCenters1_display.SelectNormalArray = 'None'
cellCenters1_display.SelectTangentArray = 'None'
cellCenters1_display.OSPRayScaleArray = 'p'
cellCenters1_display.OSPRayScaleFunction = 'PiecewiseFunction'
cellCenters1_display.SelectOrientationVectors = 'None'
cellCenters1_display.ScaleFactor = 9.95978412628174
cellCenters1_display.SelectScaleArray = 'None'
cellCenters1_display.GlyphType = 'Arrow'
cellCenters1_display.GlyphTableIndexArray = 'None'
cellCenters1_display.GaussianRadius = 0.49798920631408694
cellCenters1_display.SetScaleArray = ['POINTS', 'p']
cellCenters1_display.ScaleTransferFunction = 'PiecewiseFunction'
cellCenters1_display.OpacityArray = ['POINTS', 'p']
cellCenters1_display.OpacityTransferFunction = 'PiecewiseFunction'
cellCenters1_display.DataAxesGrid = 'GridAxesRepresentation'
cellCenters1_display.PolarAxes = 'PolarAxesRepresentation'
cellCenters1_display.SelectInputVectors = ['POINTS', '']
cellCenters1_display.WriteLog = ''

Hide(case_vtk, renderView1)
renderView1.Update()

output_path = base_dir / "CSV_data"
output_path.mkdir(parents=True, exist_ok=True)

SaveData(
    str(output_path / "CFD.csv"),
    proxy=cellCenters1,
    FieldAssociation='Point Data',
    AddMetaData=1,
    WriteTimeSteps=1,
    WriteTimeStepsSeparately=1,
    ChooseArraysToWrite=1,
    PointDataArrays=['p'],
)

print("\n--------------------\nPressure CSV job done successfully!!\n--------------------\n")
